//============================================================
// Mansion garden -> Chess Garden (Depth 2 hub)
//============================================================
ba_maison,242,122,0	script	#depth2_chess_in	WARPNPC,1,1,{
	end;
OnTouch:
	if (BaseLevel < 250) {
		mes "^0000cdThis passage is only open to those with a base level of 250 or higher.^000000";
		close;
	}
	if (isbegin_quest(16734) != 2) {
		mes "^0000cdYou must complete the Depth Biosphere introduction with the Depth Research Manager first.^000000";
		mes "^777777(ba_in01 292, 104)^000000";
		close;
	}
	warp "ba_chess", 24, 24;
	end;
}

ba_chess,24,8,0	warp2	#chess_to_maison	1,1,ba_maison,242,118

//============================================================
// Chess Garden -> Depth 2 (four spawn points)
//============================================================

	end;

function	script	F_BiosphereDepth2Enter	{
	if (BaseLevel < 250) {
		mes "^0000cdThis portal is only open to those with a base level of 250 or higher.^000000";
		close;
	}
	if (isbegin_quest(16734) != 2) {
		mes "^0000cdYou must complete the Depth Biosphere introduction with the Depth Research Manager first.^000000";
		mes "^777777(ba_in01 292, 104)^000000";
		close;
	}
	warp "bl_depth2", getarg(0), getarg(1);
	end;
}

	end;
