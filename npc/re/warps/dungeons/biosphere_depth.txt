//============================================================
// Deep Research Manager - unlock depth portals
//============================================================
ba_in01,292,105,5	script	Deep Research Manager#bs_depth	4_EP17_BASIC_B,{
	if (BaseLevel < 250) {
		mes "[Deep Research Manager]";
		mes "The depth habitats are only accessible to adventurers with base level 250 or higher.";
		close;
	}
	mes "[Deep Research Manager]";
	mes "Welcome. I manage research into the deeper specimen layers beneath the artificial habitats.";
	next;
	if (#biosphere_depth1) {
		mes "[Deep Research Manager]";
		mes "You are already registered for depth research.";
		mes "Use the depth portals inside any open habitat, or visit the Chess Garden for the abyss survey.";
		close;
	}
	mes "[Deep Research Manager]";
	mes "Would you like to register for depth habitat research?";
	mes "Once registered, depth portals in each biome will open for you.";
	next;
	if (select("Register", "Not now") == 2) {
		mes "[Deep Research Manager]";
		mes "Come back when you are ready.";
		close;
	}
	set #biosphere_depth1, 1;
	mes "[Deep Research Manager]";
	mes "Registration complete.";
	mes "Depth portals are now active in every artificial habitat.";
	close;

OnInit:
	setunittitle getnpcid(0), "<Depth Research>";
	end;
}

function	script	F_BiosphereDepth1Enter	{
	if (BaseLevel < 250) {
		mes "^0000cdThis portal is only open to those with a base level of 250 or higher.^000000";
		close;
	}
	if (!#biosphere_depth1) {
		mes "^0000cdYou must register with the Deep Research Manager in Varmundt's Mansion first.^000000";
		mes "^777777(ba_in01 292, 105)^000000";
		close;
	}
	warp "bl_depth1", 180, 165;
	end;
}
