//===== Description: =========================================
//= Material exchange, crown crafting, dimension weapon enchant.
//============================================================

// Material fusion (10 o'clock on chess board)
ba_chess,20,13,3	script	Abyss Material Manager#b2	4_EP17_TABLET,{
	if (isbegin_quest(16778) == 0) {
		mes "[Abyss Material Manager]";
		mes "Please receive the Depth Abyss briefing from the research manager first.";
		close;
	}
	mes "[Abyss Material Manager]";
	mes "I fuse abyss fragments and jewels into rune materials used for Time Dimension gear.";
	next;
	callshop "barter_bio_depth2_materials";
	end;

OnInit:
	setunittitle getnpcid(0), "<Material Exchange>";
	end;
}

// Rune crown crafting (8 o'clock)
ba_chess,21,19,3	script	Time Dimension Crown Maker#b2	4_EP17_BASKET,{
	if (isbegin_quest(16778) == 0) {
		mes "[Time Dimension Crown Maker]";
		mes "Please receive the Depth Abyss briefing from the research manager first.";
		close;
	}
	.@points = get_reputation_points(REPU_BIO_DEPTH2);
	if (.@points < 500) {
		mes "[Time Dimension Crown Maker]";
		mes "You need at least ^0000cd500^000000 Depth Abyss research reputation to craft Time Dimension Rune Crowns.";
		close;
	}
	mes "[Time Dimension Crown Maker]";
	mes "I can craft a Time Dimension Rune Crown for your advanced job.";
	next;
	callshop "barter_bio_depth2_crowns";
	end;

OnInit:
	setunittitle getnpcid(0), "<Rune Crown>";
	end;
}

// Dimension weapon enchant (2 o'clock)
ba_chess,22,16,3	script	Dimension Equipment Manager#b2	4_EP17_MASTER_A,{
	if (isbegin_quest(16778) == 0) {
		mes "[Dimension Equipment Manager]";
		mes "Please receive the Depth Abyss briefing from the research manager first.";
		close;
	}
	.@points = get_reputation_points(REPU_BIO_DEPTH2);
	mes "[Dimension Equipment Manager]";
	mes "I engrave Time Dimension weapons with abyss rune power.";
	next;
	if (.@points < 750) {
		mes "[Dimension Equipment Manager]";
		mes "You need at least ^0000cd750^000000 Depth Abyss research reputation to enchant dimension weapons.";
		close;
	}
	mes "[Dimension Equipment Manager]";
	mes "Bring the dimension weapon you wish to enchant.";
	close2;
	item_enchant(133);
	end;

OnInit:
	setunittitle getnpcid(0), "<Weapon Enchant>";
	end;
}
