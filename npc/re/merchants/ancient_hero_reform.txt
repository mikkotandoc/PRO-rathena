yuno,214,347,3	script	Zomcha#edda_ref	4_M_SAGE_C,{
	disable_items;
	mes "[Zomcha]";
	mes "I exchange surplus biolab research for reform materials.";
	mes "Use ^0000cdPatent Ancient Hero's Cube^000000 on +7 Ancient Hero gear at the Grade Market.";
	next;
	switch( select( "Exchange research materials", "About Ancient Hero upgrades", "Cancel" ) ) {
	case 1:
		mes "[Zomcha]";
		mes "Rates follow MuhRO biolab exchange values.";
		next;
		switch( select( "Back", "Patent Cube (300 Doc / 500 Frag each)", "Refine Hammer (200 Doc / 300 Frag each)", "Weapon Upgrade Stone Low (50/50)", "Weapon Upgrade Stone Mid (100/100)", "Weapon Upgrade Stone High (150/150)" ) ) {
		case 1:
			close;
		case 2:
			.@doc = 300; .@frag = 500; .@item = 100753;
			break;
		case 3:
			.@doc = 200; .@frag = 300; .@item = 100821;
			break;
		case 4:
			.@doc = 50; .@frag = 50; .@item = 1000430;
			break;
		case 5:
			.@doc = 100; .@frag = 100; .@item = 1000431;
			break;
		case 6:
			.@doc = 150; .@frag = 150; .@item = 1000432;
			break;
		}
		mes "[Zomcha]";
		mes "Trade ^0000cd" + .@doc + " Research Documents^000000 and ^0000cd" + .@frag + " Experimental Fragments^000000 for 1x " + getitemname(.@item) + ".";
		next;
		if (select( "Exchange once.", "Cancel." ) == 2)
			close;
		if (countitem(25786) < .@doc || countitem(25787) < .@frag) {
			mes "[Zomcha]";
			mes "You do not have enough research materials.";
			close;
		}
		if (!checkweight(.@item, 1)) {
			mes "[Zomcha]";
			mes "Your inventory is too full.";
			close;
		}
		delitem 25786, .@doc;
		delitem 25787, .@frag;
		getitem .@item, 1;
		mes "[Zomcha]";
		mes "Exchange complete.";
		close;
	case 2:
		mes "[Zomcha]";
		mes "^0000cdAncient Hero^000000 -> ^0000cdPatent^000000: use Patent Cube on +7 gear (refine -2).";
		mes "^0000cdPatent^000000 -> ^0000cdHero-LT^000000: see Whoopi at the Grade Market.";
		mes "EDDA Biolab weapons are enchanted at Lavian; Ancient Hero weapons come from biolab MVP boxes.";
		close;
	case 3:
		close;
	}
	end;

OnInit:
	setunittitle getnpcid(0), "<Biolab Exchange>";
	end;
}
