function	script	F_EddaBio_MvpDrops	{
	.@hard = getarg(0, 0);
	.@rate = .@hard ? 2 : 1;
	if (!checkweight(1201, 8))
		return;
	getitem 23992, 1; // Experimental Weapon Box (Test Subject Weapon Crate)
	if (rand(10000) < 50 * .@rate)
		getitem 19439, 1; // Vicious Mind Aura
	if (rand(10000) < 100 * .@rate)
		getitem 22171, 1; // Ancient Hero Boots
	if (rand(10000) < 150 * .@rate)
		getitem 100753, 1; // Patent Ancient Hero's Cube
	if (rand(10000) < 150 * .@rate)
		getitem 100821, 1; // Modified Hero's Refine Hammer
	if (rand(10000) < 200 * .@rate)
		getitem 23806, 1; // Ancient Hero Weapon Box
	if (rand(10000) < 300 * .@rate)
		getitem 23993, 1; // Bio Lab Material Box
	if (rand(10000) < 300 * .@rate)
		getitem 23994, 1; // Bio Reactor Box
	if (rand(10000) < 200 * .@rate)
		getitem 100706, 1; // Ancient Hero's Will
	return;
}

yuno,216,345,3	script	Sierra#yuno	4_F_SCIENCE,{
	if (BaseLevel < 170) {
		mes "[Sierra]";
		mes "This project is too dangerous for you right now.";
		mes "Please come back after reaching Base Level 170.";
		close;
	}
	if (checkquest(16400,PLAYTIME) == 2)
		erasequest 16400;

	mes "[Sierra]";
	mes "I'm Sierra from Secret Wings. We're investigating a hidden basement beneath the biolab.";
	mes "Will you help us relive the expedition team's memories?";
	next;
	if (select("Tell me more.", "Maybe later.") == 2) {
		mes "[Sierra]";
		mes "Come back when you're ready.";
		close;
	}
	mes "[Sierra]";
	mes "You can enter the memorial dungeon alone or with a party.";
	mes "Clearing it grants research materials used for EDDA Biolab equipment.";
	mes "^0000cdHard Mode^000000 (BL 250+) amplifies combat and improves Ancient Hero material drops.";
	next;

	switch( select( "Create Normal Instance", "Create Hard Instance", "Enter Instance", "Cancel" ) ) {
	case 1:
		.@hard = 0;
		.@name$ = .instance_name$;
		break;
	case 2:
		if (BaseLevel < 250) {
			mes "[Sierra]";
			mes "Hard Mode is reserved for adventurers with base level 250 or higher.";
			close;
		}
		.@hard = 1;
		.@name$ = .instance_hard_name$;
		break;
	case 3:
		goto L_Enter;
	case 4:
		close;
	}
	if (!is_party_leader()) {
		mes "[Sierra]";
		mes "Only the party leader can create the instance.";
		close;
	}
	switch( checkquest(16400,PLAYTIME) ) {
	case 0:
	case 1:
		mes "[Sierra]";
		mes "You still need to recover from your last expedition.";
		close;
	}
	$edda_bio_hard_pending = .@hard;
	switch( instance_create(.@name$) ) {
	case -3:
		dispbottom "Memorial Dungeon, '" + .@name$ + "' is already in progress.", 0xFFFFFF;
		close;
	case -4:
	case -2:
	case -1:
		mes "[Sierra]";
		mes "^0000FF" + .@name$ + "^000000 - instance creation failed.";
		close;
	}
	mes "[Sierra]";
	mes "The " + (.@hard ? "^FF4400hard^000000" : "standard") + " simulation is ready.";
	mes "Select Enter Instance when your party is prepared.";
	close;

L_Enter:
	switch( checkquest(16400,PLAYTIME) ) {
	case 0:
	case 1:
		mes "[Sierra]";
		mes "You still need to recover from your last expedition.";
		close;
	}
	// Normal and Hard are separate instance DB entries; try both names.
	.@enter = IE_NOINSTANCE;
	if (instance_enter(.instance_hard_name$) == IE_OK) {
		.@enter = IE_OK;
		.@announce$ = .instance_hard_name$;
		.@color$ = "0xFF8800";
	} else if (instance_enter(.instance_name$) == IE_OK) {
		.@enter = IE_OK;
		.@announce$ = .instance_name$;
		.@color$ = "0x00ff99";
	}
	switch (.@enter) {
	case IE_OK:
		setquest 16400;
		mapannounce "yuno", strcharinfo(0) + " is entering " + .@announce$ + ".", bc_map, .@color$, FW_NORMAL, 12;
		end;
	case IE_NOINSTANCE:
		mes "[Sierra]";
		mes "There is no open instance yet. Create one first.";
		close;
	default:
		mes "[Sierra]";
		mes "An unknown error has occurred.";
		mes "If a memorial dungeon is already reserved, use ^0000cdDestroy instance^000000 in the Memorial Dungeon window and create a new one.";
		close;
	}
	end;

OnInit:
	set .instance_name$, "EDDA Somatology Laboratory";
	set .instance_hard_name$, "EDDA Somatology Laboratory (Hard)";
	setunittitle getnpcid(0), "<Bioresearch Lab>";
	end;
}
