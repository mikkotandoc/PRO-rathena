function	script	F_OGHCM_MaxLevel	{
	.@clears = #OGHCM_Clear;
	if (.@clears >= 100) return 10;
	if (.@clears >= 80) return 9;
	if (.@clears >= 60) return 8;
	if (.@clears >= 45) return 7;
	if (.@clears >= 35) return 6;
	if (.@clears >= 26) return 5;
	if (.@clears >= 18) return 4;
	if (.@clears >= 11) return 3;
	if (.@clears >= 5) return 2;
	return 1;
}

function	script	F_OGHCM_Reward	{
	.@level = getarg(0);
	.@bonus_root = getarg(1,0);
	.@bonus_slave = getarg(2,0);
	.@bonus_clean = getarg(3,0);

	// Oscar performance bonus (separate from the boss treasure chest).
	switch (.@level) {
	case 1: .@c_min = 1; .@c_max = 2; break;
	case 2: .@c_min = 1; .@c_max = 2; break;
	case 3: .@c_min = 2; .@c_max = 3; break;
	case 4: .@c_min = 2; .@c_max = 3; break;
	case 5: .@c_min = 2; .@c_max = 4; break;
	default: .@c_min = 3; .@c_max = 5; break;
	}
	if (.@bonus_root) .@c_max++;
	if (.@bonus_slave) .@c_max++;
	if (.@bonus_clean) .@c_max++;
	getitem 25864, rand(.@c_min, min(.@c_max, 10)); // S_Circlet_Of_Time
	return;
}

function	script	F_OGHCM_MobDrop	{
	if (rand(10000) <= 100) {
		setarray .@weapon[0],
			1336, 1870, 2060, 13347, 18191, 18198, 21055, 26165, 26166, 26172,
			26216, 28046, 28141, 28636, 28774, 28775, 28776, 32027, 32111, 32304,
			32353, 32401, 32402, 32403;
		.@idx = rand(getarraysize(.@weapon));
		if (checkweight(.@weapon[.@idx], 1))
			getitem .@weapon[.@idx], 1;
		return;
	}
	if (rand(100) <= 35 && checkweight(25867, 3))
		getitem 25867, rand(1, 3); // Fragment_Of_Time
	return;
}

function	script	F_OGHCM_MobScale	{
	.@gid = getarg(0);
	.@level = getarg(1, 1);
	if (.@gid <= 0 || .@level < 1)
		return;
	getunitdata .@gid, .@u;
	setunitdata .@gid, UMOB_ATKMIN, max(1, .@u[UMOB_ATKMIN] * .@level);
	setunitdata .@gid, UMOB_ATKMAX, max(1, .@u[UMOB_ATKMAX] * .@level);
	setunitdata .@gid, UMOB_MATKMIN, max(1, .@u[UMOB_MATKMIN] * .@level);
	setunitdata .@gid, UMOB_MATKMAX, max(1, .@u[UMOB_MATKMAX] * .@level);
	return;
}

function	script	F_OGHCM_ThornScale	{
	.@gid = getarg(0);
	.@level = getarg(1, 1);
	if (.@gid <= 0 || .@level < 1)
		return;
	.@hp = (5 + (.@level * 4)) * 10000;
	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP, .@hp;
	return;
}

function	script	F_OGHCM_CountLivingArena	{
	.@party_id = getarg(0);
	.@map$ = getarg(1);
	.@alive = 0;
	.@back_aid = getcharid(0);
	getpartymember .@party_id, 1;
	getpartymember .@party_id, 2;
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		if (isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]) == 0)
			continue;
		if (attachrid($@partymemberaid[.@i]) == 0)
			continue;
		if (strcharinfo(3) == .@map$ && Hp > 0)
			.@alive++;
	}
	if (.@back_aid)
		attachrid(.@back_aid);
	return .@alive;
}

function	script	F_OGHCM_ApplyBossPenalty	{
	.@gid = getarg(0);
	.@active = getarg(1);
	if (.@gid <= 0)
		return;
	getunitdata .@gid, .@u;
	if (.@active) {
		if (!'OGHCM_BossPenalty) {
			'OGHCM_BossBaseDef = .@u[UMOB_DEF];
			'OGHCM_BossBaseMDef = .@u[UMOB_MDEF];
			setunitdata .@gid, UMOB_DEF, .@u[UMOB_DEF] * 1000;
			setunitdata .@gid, UMOB_MDEF, .@u[UMOB_MDEF] * 1000;
			'OGHCM_BossPenalty = 1;
		}
	} else if ('OGHCM_BossPenalty) {
		setunitdata .@gid, UMOB_DEF, 'OGHCM_BossBaseDef;
		setunitdata .@gid, UMOB_MDEF, 'OGHCM_BossBaseMDef;
		'OGHCM_BossPenalty = 0;
	}
	return;
}

function	script	F_OGHCM_BossScale	{
	.@gid = getarg(0);
	.@level = getarg(1);
	if (.@gid <= 0 || .@level < 1)
		return;
	getunitdata .@gid, .@u;
	.@maxhp = callfunc("F_OGHCM_BossHP", .@level);
	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	setunitdata .@gid, UMOB_ATKMIN, max(1, .@u[UMOB_ATKMIN] * .@level);
	setunitdata .@gid, UMOB_ATKMAX, max(1, .@u[UMOB_ATKMAX] * .@level);
	setunitdata .@gid, UMOB_MATKMIN, max(1, .@u[UMOB_MATKMIN] * .@level);
	setunitdata .@gid, UMOB_MATKMAX, max(1, .@u[UMOB_MATKMAX] * .@level);
	setunitdata .@gid, UMOB_DEF, .@u[UMOB_DEF] + ((.@level - 1) * 50);
	setunitdata .@gid, UMOB_MDEF, .@u[UMOB_MDEF] + ((.@level - 1) * 10);
	'OGHCM_BossBaseDef = .@u[UMOB_DEF] + ((.@level - 1) * 50);
	'OGHCM_BossBaseMDef = .@u[UMOB_MDEF] + ((.@level - 1) * 10);
	'OGHCM_BossPenalty = 0;
	return;
}

// Returns 0 = ok, 1 = self on cooldown, 2 = party leader on cooldown.
function	script	F_OGHCM_CheckEntryCooldown	{
	.@party_id = getarg(0);

	switch( checkquest(12493,PLAYTIME) ) {
	case 0:
	case 1:
		return 1;
	case 2:
		if (isbegin_quest(12493) > 0)
			erasequest 12493;
		break;
	}

	if (.@party_id > 0) {
		.@leader_cid = getpartyleader(.@party_id, 2);
		if (.@leader_cid > 0 && .@leader_cid != getcharid(0)) {
			.@leader_aid = getpartyleader(.@party_id, 1);
			if (.@leader_aid > 0 && isloggedin(.@leader_aid, .@leader_cid)) {
				switch( checkquest(12493,PLAYTIME, .@leader_cid) ) {
				case 0:
				case 1:
					return 2;
				case 2:
					if (isbegin_quest(12493, .@leader_cid) > 0)
						erasequest 12493, .@leader_cid;
					break;
				}
			}
		}
	}
	return 0;
}

function	script	F_OGHCM_ApplyEntryQuest	{
	if (isbegin_quest(12493) == 0)
		setquest 12493;
	return;
}

function	script	F_OGHCM_InstanceExit	{
	.@party_id = getarg(0);
	getpartymember .@party_id, 1;
	getpartymember .@party_id, 2;
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		if (isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]) == 0)
			continue;
		warpchar "glast_01", 139, 283, $@partymembercid[.@i];
	}
	instance_destroy;
	return;
}

function	script	F_OGHCM_ApplyClearCooldown	{
	if (isbegin_quest(12493) == 0)
		setquest 12493;
	else if (checkquest(12493,PLAYTIME) == 2) {
		erasequest 12493;
		setquest 12493;
	}
	completequest 12493;
	return;
}

		close;

		mes "Shall I send you outside?";
		next;

			close;
		close2;
		callfunc("F_OGHCM_InstanceExit", 'party_id);
		end;
	}
	mes "[Oscar]";
	mes "Well done. Here are your performance rewards for Contamination Level " + 'OGHCM_Level + ".";
	callfunc("F_OGHCM_Reward", 'OGHCM_Level, 'OGHCM_RootKilled, ('OGHCM_SlaveKills >= 100), 'OGHCM_BossClean);
	if ('OGHCM_BossIsHimmel)
		getitem 102553, 1; // C_Himel_Antiquity
	else
		getitem 102552, 1; // C_Amdarais_Antiquity
	#OGHCM_Clear++;
	callfunc("F_OGHCM_ApplyClearCooldown");
	@oghcm_reward = 1;
	mes "[Oscar]";
	mes "Total clears: ^0000CD" + #OGHCM_Clear + "^000000";
	close2;
	callfunc("F_OGHCM_InstanceExit", 'party_id);
	end;
}
