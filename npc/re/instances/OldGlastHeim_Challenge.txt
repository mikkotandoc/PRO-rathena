function	script	F_OGHCM_MaxLevel	{
	.@clears = #OGHCM_Clear;
	if (.@clears >= 100) return 10;
	if (.@clears >= 80) return 9;
	if (.@clears >= 60) return 8;
	if (.@clears >= 45) return 7;
	if (.@clears >= 35) return 6;
	if (.@clears >= 26) return 5;
	if (.@clears >= 18) return 4;
	if (.@clears >= 11) return 3;
	if (.@clears >= 5) return 2;
	return 1;
}

function	script	F_OGHCM_Reward	{
	.@level = getarg(0);
	.@bonus_root = getarg(1,0);
	.@bonus_slave = getarg(2,0);
	.@bonus_clean = getarg(3,0);

	// Oscar performance bonus (separate from the boss treasure chest).
	switch (.@level) {
	case 1: .@c_min = 1; .@c_max = 2; break;
	case 2: .@c_min = 1; .@c_max = 2; break;
	case 3: .@c_min = 2; .@c_max = 3; break;
	case 4: .@c_min = 2; .@c_max = 3; break;
	case 5: .@c_min = 2; .@c_max = 4; break;
	default: .@c_min = 3; .@c_max = 5; break;
	}
	if (.@bonus_root) .@c_max++;
	if (.@bonus_slave) .@c_max++;
	if (.@bonus_clean) .@c_max++;
	getitem 25864, rand(.@c_min, min(.@c_max, 10)); // S_Circlet_Of_Time
	return;
}

function	script	F_OGHCM_MobDrop	{
	if (rand(10000) <= 100) {
		setarray .@weapon[0],
			1336, 1870, 2060, 13347, 18191, 18198, 21055, 26165, 26166, 26172,
			26216, 28046, 28141, 28636, 28774, 28775, 28776, 32027, 32111, 32304,
			32353, 32401, 32402, 32403;
		.@idx = rand(getarraysize(.@weapon));
		if (checkweight(.@weapon[.@idx], 1))
			getitem .@weapon[.@idx], 1;
		return;
	}
	if (rand(100) <= 35 && checkweight(25867, 3))
		getitem 25867, rand(1, 3); // Fragment_Of_Time
	return;
}

function	script	F_OGHCM_RootDrop	{
	if (checkweight(25865, 2) == 0)
		return;
	if (rand(100) <= 80)
		getitem 25865, rand(1, 2); // Jewel_Of_Time_Ore
	if (rand(100) <= 40)
		getitem 25867, rand(1, 3);
	return;
}

function	script	F_OGHCM_BossScale	{
	.@gid = getarg(0);
	.@level = getarg(1);
	if (.@gid <= 0)
		return;
	.@maxhp = 600000000 + ((.@level - 1) * 150000000);
	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	return;
}

function	script	F_OGHCM_ShowNpc	{
	.@name$ = getarg(0);
	enablenpc .@name$;
	cloakoffnpc .@name$;
	return;
}

// Returns 0 = ok, 1 = self on cooldown, 2 = party leader on cooldown.
function	script	F_OGHCM_CheckEntryCooldown	{
	.@party_id = getarg(0);

	switch( checkquest(12493,PLAYTIME) ) {
	case 0:
	case 1:
		return 1;
	case 2:
		if (isbegin_quest(12493) > 0)
			erasequest 12493;
		break;
	}

	if (.@party_id > 0) {
		.@leader_cid = getpartyleader(.@party_id, 2);
		if (.@leader_cid > 0 && .@leader_cid != getcharid(0)) {
			.@leader_aid = getpartyleader(.@party_id, 1);
			if (.@leader_aid > 0 && isloggedin(.@leader_aid, .@leader_cid)) {
				switch( checkquest(12493,PLAYTIME, .@leader_cid) ) {
				case 0:
				case 1:
					return 2;
				case 2:
					if (isbegin_quest(12493, .@leader_cid) > 0)
						erasequest 12493, .@leader_cid;
					break;
				}
			}
		}
	}
	return 0;
}

function	script	F_OGHCM_ApplyEntryQuest	{
	if (isbegin_quest(12493) == 0)
		setquest 12493;
	return;
}

function	script	F_OGHCM_ApplyClearCooldown	{
	if (isbegin_quest(12493) == 0)
		setquest 12493;
	else if (checkquest(12493,PLAYTIME) == 2) {
		erasequest 12493;
		setquest 12493;
	}
	completequest 12493;
	return;
}

			'OGHCM_GridExpire = gettimetick(2) + 5;
		}
		if ('OGHCM_GridExpire > 0 && gettimetick(2) >= 'OGHCM_GridExpire) {
			killmonster 'map2$, instance_npcname("OGHCM Controller") + "::OnGridDead";
			'OGHCM_GridExpire = 0;
		}
		if (.@pct <= 70 && !'OGHCM_Phase70) {
			'OGHCM_Phase70 = 1;
			mapannounce 'map2$, "Oscar: Mutated Khalitzburgs join the battle!", bc_map, 0xFFFF00;
			for (.@i = 0; .@i < 4 + 'OGHCM_Level; .@i++)
				monster 'map2$, 140 + rand(20), 50 + rand(20), "--ja--", "MD_C_KHALITZBURG", 1, instance_npcname("OGHCM Controller") + "::OnSlaveDead";
		}
		if (.@pct <= 50 && !'OGHCM_Phase50) {
			'OGHCM_Phase50 = 1;
			mapannounce 'map2$, "Oscar: Mutated White Knights charge in!", bc_map, 0xFFFF00;
			for (.@i = 0; .@i < 3 + 'OGHCM_Level; .@i++)
				monster 'map2$, 140 + rand(20), 50 + rand(20), "--ja--", "MD_C_WHITEKNIGHT", 1, instance_npcname("OGHCM Controller") + "::OnSlaveDead";
		}
		if (.@pct <= 30 && !'OGHCM_Phase30) {
			'OGHCM_Phase30 = 1;
			mapannounce 'map2$, "Oscar: Mutated Raydric Archers surround the arena!", bc_map, 0xFFFF00;
			for (.@i = 0; .@i < 3 + 'OGHCM_Level; .@i++)
				monster 'map2$, 140 + rand(20), 50 + rand(20), "--ja--", "MD_C_RAYDRIC_ARCHER", 1, instance_npcname("OGHCM Controller") + "::OnSlaveDead";
		}
		.@regen = 300000 * 'OGHCM_Level;
		if (.@hp < .@maxhp)
			setunitdata 'OGHCM_BossGid, UMOB_HP, min(.@maxhp, .@hp + .@regen);
		if (distance(150, 60, .@u[UMOB_X], .@u[UMOB_Y]) > 12 && !'OGHCM_ArenaWarn) {
			'OGHCM_ArenaWarn = 1;
			'OGHCM_ArenaExpire = gettimetick(2) + 30;
			mapannounce 'map2$, "Oscar: Bring the phantom back to the center within 30 seconds!", bc_map, 0xFF0000;
		} else if (distance(150, 60, .@u[UMOB_X], .@u[UMOB_Y]) <= 12 && 'OGHCM_ArenaWarn) {
			'OGHCM_ArenaWarn = 0;
			'OGHCM_ArenaExpire = 0;
		}
		if ('OGHCM_ArenaExpire > 0 && gettimetick(2) >= 'OGHCM_ArenaExpire) {
			mapannounce 'map2$, "Oscar: The phantom fled the arena. The battle restarts!", bc_map, 0xFF0000;
			killmonster 'map2$, instance_npcname("OGHCM Controller") + "::OnSlaveDead";
			killmonster 'map2$, instance_npcname("OGHCM Controller") + "::OnGridDead";
			donpcevent instance_npcname("OGHCM Controller") + "::OnBossEscape";
			end;
		}
		sleep 2000;
	}
	end;
}
