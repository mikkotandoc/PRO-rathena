//===== rAthena Script =======================================
//= Episode 21 memorial dungeons
//===== Description: =========================================
//= Playable Episode 21 instance framework for Gimle,
//= Mysterious Ghost Ship, Secret Altar, Final Battle, and
//= Silent Sanctuary. Access is level 230+, party-based, and
//= protected by a 20-hour account cooldown.
//============================================================

function	script	F_ep21_instance_access	{
	.@name$ = getarg(0);
	.@index = getarg(1);
	.@party_id = getcharid(1);

	if (BaseLevel < 230) {
		mes "Only adventurers of Base Level 230 or higher may enter.";
		return;
	}
	if (.@party_id < 1) {
		mes "Create a party before requesting entry.";
		return;
	}
	if (#ep21_instance_cd[.@index] > gettimetick(2)) {
		.@remaining = #ep21_instance_cd[.@index] - gettimetick(2);
		mes "Your account is still recovering from this mission.";
		mes "Remaining time: " + (.@remaining / 3600) + " hour(s), " + ((.@remaining % 3600) / 60) + " minute(s).";
		return;
	}

	if (instance_live_info(ILI_NAME,instance_id(IM_PARTY)) == .@name$) {
		if (select("Enter " + .@name$ + ":Cancel") == 2)
			return;
		switch (instance_enter(.@name$)) {
		case IE_OK:
			#ep21_instance_cd[.@index] = gettimetick(2) + 72000;
			end;
		case IE_NOMEMBER:
			mes "You are not a member of the party that created this instance.";
			return;
		case IE_NOINSTANCE:
			mes "The dimensional passage is not ready.";
			return;
		default:
			mes "The instance could not be entered.";
			return;
		}
	}

	if (!is_party_leader()) {
		mes "Only the party leader can create this instance.";
		return;
	}
	if (select("Create " + .@name$ + ":Cancel") == 2)
		return;
	if (instance_create(.@name$) < 0) {
		mes "The memorial dungeon could not be created.";
		return;
	}
	mes "The dimensional passage has opened.";
	mes "Talk to me again to enter.";
	return;
}

function	script	F_ep21_instance_reward	{
	.@hard = getarg(0,0);
	.@aid = getcharid(3);

	for (.@i = 0; .@i < getarraysize('ep21_reward_aid); ++.@i) {
		if ('ep21_reward_aid[.@i] == .@aid) {
			dispbottom "You already claimed this instance reward.";
			return;
		}
	}
	'ep21_reward_aid[getarraysize('ep21_reward_aid)] = .@aid;
	if (.@hard) {
		getexp 100000000,70000000;
		Zeny += 1000000;
	} else {
		getexp 50000000,35000000;
		Zeny += 500000;
	}
	dispbottom "Episode 21 clear reward received.";
	return;
}
