function	script	F_lake_energy_id	{
	switch( getarg(0) ) {
	case 22177: return 1001440;
	case 22178: return 1001441;
	case 22179: return 1001442;
	case 22180: return 1001443;
	}
	return 1001440;
}

function	script	F_lake_boss_name$	{
	switch( getarg(0) ) {
	case 22177: return "Floral Pegasus";
	case 22178: return "Coral Marine";
	case 22179: return "Maple Tree";
	case 22180: return "Monarch Penguin";
	}
	return "Divine Beast";
}

function	script	F_lake_boss_season$	{
	switch( getarg(0) ) {
	case 22177: return "Spring";
	case 22178: return "Summer";
	case 22179: return "Autumn";
	case 22180: return "Winter";
	}
	return "Season";
}

function	script	F_lake_hunt_quest	{
	switch( getarg(0) ) {
	case 22177: return 12613;
	case 22178: return 12614;
	case 22179: return 12615;
	case 22180: return 12616;
	}
	return 0;
}

function	script	F_lake_complete_hunt	{
	.@q = callfunc("F_lake_hunt_quest", getarg(0));
	if (.@q < 1)
		return;
	if (checkquest(.@q,HUNTING) == 2)
		return;
	if (checkquest(.@q) == -1)
		setquest .@q;
	completequest .@q;
	return;
}

function	script	F_lake_account_rewarded	{
	.@aid = getarg(0);
	for ( .@i = 0; .@i < getarraysize('lake_reward_aid); ++.@i ) {
		if ('lake_reward_aid[.@i] == .@aid)
			return 1;
	}
	return 0;
}

function	script	F_lake_give_reward	{
	.@aid = getcharid(3);
	.@energy = callfunc("F_lake_energy_id", 'lake_boss_id);

	if (!callfunc("F_lake_account_rewarded", .@aid)) {
		'lake_reward_aid[getarraysize('lake_reward_aid)] = .@aid;
		getitem 1001415,1; // Hall of Life Key
		getitem 1001414,30; // Dimensional Crystal Fragment
		getitem .@energy,15;
		dispbottom "Lake of Fire: full clear reward granted (first character on this master account).";
	} else {
		getitem 1001414,10;
		getitem .@energy,5;
		dispbottom "Lake of Fire: reduced reward granted (additional character on this master account).";
	}
	callfunc("F_lake_complete_hunt", 'lake_boss_id);
	return;
}

function	script	F_lake_set_protection	{
	if (!unitexists('lake_boss_gid))
		return;
	switch( getarg(0) ) {
	case 0: // Phase 2 — barrier broken, 10% damage taken
		// UMOB_DAMAGETAKEN is percent taken (cap at 99). Values other than 100
		// are scaled, then floored to at least 1 HP (see battle_apply_mob_damagetaken).
		setunitdata 'lake_boss_gid, UMOB_DAMAGETAKEN, 10;
		setunitdata 'lake_boss_gid, UMOB_DMGIMMUNE, 0;
		break;
	case 1: // Lure — 1 HP per hit + HP lock (damagetaken 0 => floor to 1)
		setunitdata 'lake_boss_gid, UMOB_DAMAGETAKEN, 0;
		setunitdata 'lake_boss_gid, UMOB_DMGIMMUNE, 0;
		callfunc("F_lake_lock_hp");
		break;
	case 2: // Brief trap stun — still lure barrier (1 HP/hit); HP lock allows limited chip
		setunitdata 'lake_boss_gid, UMOB_DAMAGETAKEN, 0;
		setunitdata 'lake_boss_gid, UMOB_DMGIMMUNE, 0;
		break;
	}
	return;
}

function	script	F_lake_lock_hp	{
	if ('lake_phase != 1 || !unitexists('lake_boss_gid))
		return;
	getunitdata 'lake_boss_gid, .@b;
	if (.@b[UMOB_HP] < 'lake_boss_hp_lock)
		setunitdata 'lake_boss_gid, UMOB_HP, 'lake_boss_hp_lock;
	if (.@b[UMOB_DAMAGETAKEN] != 0 && 'lake_trap_window < gettimetick(2))
		setunitdata 'lake_boss_gid, UMOB_DAMAGETAKEN, 0;
	return;
}

function	script	F_lake_init_boss	{
	.@event$ = getarg(0, "");
	// monster() stores GID in $@mobid[] — it does not return a value
	monster 'lake_map$, 'lake_boss_x, 'lake_boss_y, "--ja--", 'lake_boss_id, 1, .@event$;
	'lake_boss_gid = $@mobid[0];
	if ('lake_boss_gid <= 0)
		return 0;
	setunitdata 'lake_boss_gid, UMOB_MODE, MD_CANMOVE|MD_CANATTACK;
	setunitdata 'lake_boss_gid, UMOB_MAXHP, 700000000;
	setunitdata 'lake_boss_gid, UMOB_HP, 700000000;
	'lake_boss_hp_lock = 700000000;
	callfunc("F_lake_set_protection", 1);
	return 'lake_boss_gid;
}

function	script	F_lake_start_lure	{
	if ('lake_started || !unitexists('lake_boss_gid))
		return 0;
	'lake_started = 1;
	setunitdata 'lake_boss_gid, UMOB_MODE, MD_CANMOVE|MD_AGGRESSIVE|MD_ANGRY|MD_CANATTACK;
	callfunc("F_lake_set_protection", 1);
	donpcevent instance_npcname("#lake_boss_ai") + "::OnStart";
	donpcevent instance_npcname("#lake_trap_ai") + "::OnStart";
	mapannounce 'lake_map$, callfunc("F_lake_boss_name$", 'lake_boss_id) + " is protected by a dimensional barrier. Lure it onto the eight seals! (0/8)", bc_map, 0xFFDD44;
	return 1;
}

function	script	F_lake_mark_players_in_square	{
	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	.@size = getarg(3, 2);
	getareaunits(BL_PC, .@map$, .@x - .@size, .@y - .@size, .@x + .@size, .@y + .@size, .@pc[0]);
	for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
		if (!attachrid(.@pc[.@i]))
			continue;
		@lake_marked = gettimetick(2) + 4;
		detachrid;
	}
	return;
}

function	script	F_lake_damage_marked	{
	.@pct = getarg(0, 20);
	getareaunits(BL_PC, 'lake_map$, 0, 0, 150, 200, .@pc[0]);
	for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
		if (!attachrid(.@pc[.@i]))
			continue;
		if (@lake_marked >= gettimetick(2))
			percentheal -.@pct, 0;
		@lake_marked = 0;
		detachrid;
	}
	return;
}

function	script	F_lake_cast_small_fields	{
	.@skill$ = getarg(0, "WZ_METEOR");
	.@announce$ = getarg(1, "Danger zones are forming!");
	getunitdata 'lake_boss_gid, .@b;
	mapannounce 'lake_map$, .@announce$, bc_map, 0xFF8800;
	for ( .@f = 0; .@f < 3; ++.@f ) {
		'lake_field_x[.@f] = .@b[UMOB_X] + rand(-8, 8);
		'lake_field_y[.@f] = .@b[UMOB_Y] + rand(-8, 8);
		npcskilleffect .@skill$, 1, 'lake_field_x[.@f], 'lake_field_y[.@f];
	}
	sleep 2000;
	for ( .@f = 0; .@f < 3; ++.@f )
		callfunc("F_lake_mark_players_in_square", 'lake_map$, 'lake_field_x[.@f], 'lake_field_y[.@f], 2);
	mapannounce 'lake_map$, "Marked targets will be struck soon!", bc_map, 0xFF4400;
	sleep 2000;
	callfunc("F_lake_damage_marked", getarg(2, 20));
	return;
}

function	script	F_lake_cast_block_explosion	{
	getunitdata 'lake_boss_gid, .@b;
	mapannounce 'lake_map$, "Coral Marine is preparing Block Explosion! Use an Unsealing Scroll to break free.", bc_map, 0x00CCFF;
	for ( .@f = 0; .@f < 3; ++.@f ) {
		'lake_field_x[.@f] = .@b[UMOB_X] + rand(-8, 8);
		'lake_field_y[.@f] = .@b[UMOB_Y] + rand(-8, 8);
		npcskilleffect "WZ_WATERBALL", 5, 'lake_field_x[.@f], 'lake_field_y[.@f];
	}
	sleep 2000;
	for ( .@f = 0; .@f < 3; ++.@f )
		callfunc("F_lake_mark_players_in_square", 'lake_map$, 'lake_field_x[.@f], 'lake_field_y[.@f], 2);
	sleep 1500;
	getareaunits(BL_PC, 'lake_map$, 0, 0, 150, 200, .@pc[0]);
	for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
		if (!attachrid(.@pc[.@i]))
			continue;
		if (@lake_marked >= gettimetick(2)) {
			sc_start SC_WHITEIMPRISON, 6000, 1;
			percentheal -15, 0;
		}
		@lake_marked = 0;
		detachrid;
	}
	return;
}

function	script	F_lake_cast_frost_field	{
	getunitdata 'lake_boss_gid, .@b;
	.@x = .@b[UMOB_X];
	.@y = .@b[UMOB_Y];
	mapannounce 'lake_map$, "Monarch Penguin creates a Frost Field! Warm Shield Scrolls block the chill.", bc_map, 0x99CCFF;
	npcskilleffect "WZ_FROSTNOVA", 10, .@x, .@y;
	sleep 2000;
	getareaunits(BL_PC, 'lake_map$, .@x - 7, .@y - 7, .@x + 7, .@y + 7, .@pc[0]);
	for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
		if (!attachrid(.@pc[.@i]))
			continue;
		if (@WARM_SHIELD > gettimetick(2)) {
			detachrid;
			continue;
		}
		percentheal -18, 0;
		@lake_frosted = 1;
		sc_start SC_FREEZE, 4000, 0;
		detachrid;
	}
	sleep 1500;
	mapannounce 'lake_map$, "Lightning Judgement!", bc_map, 0xFFFF44;
	npcskilleffect "WZ_VERMILION", 10, .@x, .@y;
	getareaunits(BL_PC, 'lake_map$, .@x - 7, .@y - 7, .@x + 7, .@y + 7, .@pc[0]);
	for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
		if (!attachrid(.@pc[.@i]))
			continue;
		if (@lake_frosted && @WARM_SHIELD <= gettimetick(2))
			percentheal -25, 0;
		@lake_frosted = 0;
		detachrid;
	}
	return;
}

1@f_lake,0,0,0	script	LakeTrap	-1,{
	end;
OnTouch:
	if ('lake_phase != 1 || !'lake_started)
		end;
	getmapxy .@m$, .@x, .@y, BL_NPC;
	.@idx = callsub(S_TrapIndex, .@x, .@y);
	if (.@idx < 1)
		end;
	if ('lake_trap_done[.@idx]) {
		dispbottom "This dimensional seal has already been activated.";
		end;
	}
	if (!unitexists('lake_boss_gid)) {
		npctalk "The divine beast must stand on this seal.", "", bc_self;
		end;
	}
	getunitdata 'lake_boss_gid, .@b;
	if (distance(.@b[UMOB_X], .@b[UMOB_Y], .@x, .@y) > 14) {
		npctalk "The seal stirs... lure the divine beast closer!", "", bc_self;
		dispbottom "Lure the divine beast onto this seal (within 14 cells).";
		end;
	}
	'lake_trap_pending = .@idx;
	donpcevent instance_npcname("#lake_trap_ai") + "::OnActivate";
	end;
OnTouchNPC:
	if ('lake_phase != 1 || !'lake_started)
		end;
	if (getattachedrid() != 'lake_boss_gid)
		end;
	getmapxy .@m$, .@x, .@y, BL_NPC;
	.@idx = callsub(S_TrapIndex, .@x, .@y);
	if (.@idx < 1 || 'lake_trap_done[.@idx])
		end;
	'lake_trap_pending = .@idx;
	donpcevent instance_npcname("#lake_trap_ai") + "::OnActivate";
	end;
S_TrapIndex:
	.@x = getarg(0);
	.@y = getarg(1);
	for ( .@t = 1; .@t <= 8; ++.@t ) {
		if (distance(.@x, .@y, 'lake_trap_x[.@t], 'lake_trap_y[.@t]) <= 4)
			return .@t;
	}
	return 0;
}

1@f_lake,1,1,0	script	#lake_boss_ai	-1,{
	end;
OnStart:
	'lake_boss_tick = 0;
	initnpctimer;
	end;
OnStop:
	stopnpctimer;
	end;
OnTimer200:
	if ('lake_phase == 1 && 'lake_started && unitexists('lake_boss_gid)) {
		getunitdata 'lake_boss_gid, .@b;
		if ('lake_trap_window < gettimetick(2)) {
			if (.@b[UMOB_HP] < 'lake_boss_hp_lock)
				setunitdata 'lake_boss_gid, UMOB_HP, 'lake_boss_hp_lock;
			if (.@b[UMOB_DAMAGETAKEN] != 0)
				setunitdata 'lake_boss_gid, UMOB_DAMAGETAKEN, 0;
		} else {
			// After a trap: keep HP lock baseline but allow chip damage during stun window
			if (.@b[UMOB_HP] < 'lake_boss_hp_lock - 50000000)
				setunitdata 'lake_boss_gid, UMOB_HP, 'lake_boss_hp_lock - 50000000;
		}
		'lake_boss_tick++;
		if ('lake_boss_tick % 3 == 0 && 'lake_trap_window < gettimetick(2)) {
			getunitdata 'lake_boss_gid, .@b;
			getareaunits(BL_PC, 'lake_map$, 0, 0, 150, 200, .@pc[0]);
			.@best = 9999;
			.@tx = .@b[UMOB_X];
			.@ty = .@b[UMOB_Y];
			for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
				getmapxy .@m$, .@px, .@py, BL_PC, .@pc[.@i];
				if (.@m$ != 'lake_map$)
					continue;
				.@d = distance(.@px, .@py, .@b[UMOB_X], .@b[UMOB_Y]);
				if (.@d < .@best) {
					.@best = .@d;
					.@tx = .@px;
					.@ty = .@py;
				}
			}
			if (.@best < 9999) {
				.@dx = (.@tx > .@b[UMOB_X] ? 1 : (.@tx < .@b[UMOB_X] ? -1 : 0));
				.@dy = (.@ty > .@b[UMOB_Y] ? 1 : (.@ty < .@b[UMOB_Y] ? -1 : 0));
				unitwalk 'lake_boss_gid, .@b[UMOB_X] + (.@dx * 3), .@b[UMOB_Y] + (.@dy * 3);
			}
		}
		// God's Shadow (lure phase)
		if ('lake_boss_tick % 40 == 0) {
			getunitdata 'lake_boss_gid, .@b;
			mapannounce 'lake_map$, "God's Shadow!", bc_map, 0xFF8800;
			npcskilleffect "WZ_JUPITEL", 5, .@b[UMOB_X], .@b[UMOB_Y];
			getareaunits(BL_PC, 'lake_map$, .@b[UMOB_X]-12, .@b[UMOB_Y]-12, .@b[UMOB_X]+12, .@b[UMOB_Y]+12, .@hit[0]);
			for ( .@j = 0; .@j < getarraysize(.@hit); ++.@j ) {
				if (!attachrid(.@hit[.@j]))
					continue;
				percentheal -8, 0;
				pushpc rand(8), 3;
				detachrid;
			}
		}
	}
	if ('lake_phase == 1 && 'lake_started) {
		initnpctimer;
		setnpctimer 200;
	}
	end;
}

1@f_lake,1,1,0	script	#lake_trap_ai	-1,{
	end;
OnStart:
	initnpctimer;
	end;
OnStop:
	stopnpctimer;
	end;
OnActivate:
	callsub(S_DoTrap, 'lake_trap_pending);
	end;
OnTimer200:
	if ('lake_phase != 1 || !'lake_started) {
		initnpctimer;
		end;
	}
	if (unitexists('lake_boss_gid)) {
		getunitdata 'lake_boss_gid, .@b;
		for ( .@t = 1; .@t <= 8; ++.@t ) {
			if ('lake_trap_done[.@t])
				continue;
			if (distance(.@b[UMOB_X], .@b[UMOB_Y], 'lake_trap_x[.@t], 'lake_trap_y[.@t]) <= 10) {
				callsub(S_DoTrap, .@t);
				initnpctimer;
				end;
			}
		}
	}
	initnpctimer;
	end;
S_DoTrap:
	.@idx = getarg(0);
	if (!'lake_started || 'lake_phase != 1)
		return;
	if (.@idx < 1 || .@idx > 8 || 'lake_trap_done[.@idx])
		return;
	'lake_trap_done[.@idx] = 1;
	npcskilleffect "WZ_VERMILION", 10, 'lake_trap_x[.@idx], 'lake_trap_y[.@idx];
	npcskilleffect "MG_SAFETYWALL", 10, 'lake_trap_x[.@idx], 'lake_trap_y[.@idx];
	disablenpc instance_npcname("LakeTrap#" + .@idx);
	if (unitexists('lake_boss_gid)) {
		sc_start SC_STUN, 3500, 0, 10000, 0, 'lake_boss_gid;
		'lake_trap_window = gettimetick(2) + 4;
		callfunc("F_lake_set_protection", 2);
		getunitdata 'lake_boss_gid, .@b;
		'lake_boss_hp_lock = .@b[UMOB_HP];
		unittalk 'lake_boss_gid, "Urgh... a seal cracks!";
	}
	'lake_traps++;
	mapannounce 'lake_map$, "Dimensional seal activated! (" + 'lake_traps + "/8) — strike while it is stunned!", bc_map, 0x00FF99;
	if ('lake_traps >= 8)
		donpcevent instance_npcname("#lake_of_fire_ctrl") + "::OnTrapReady";
	return;
}

1@f_lake,1,1,0	script	#lake_skill_ai	-1,{
