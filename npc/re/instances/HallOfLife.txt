function	script	F_md_hol_get_level	{
	if (#MD_HOL_LEVEL < 1)
		#MD_HOL_LEVEL = 1;
	if (#MD_HOL_LEVEL > 20)
		#MD_HOL_LEVEL = 20;
	return #MD_HOL_LEVEL;
}

function	script	F_md_hol_sync_card	{
	.@lv = callfunc("F_md_hol_get_level");
	.@card = 312452 + .@lv; // MD_Hol_Barrier_1 .. _20 (312453-312472)
	if (isequippedcnt(420231) > 0) {
		delequip EQI_HEAD_LOW;
		getitem2 420231,1,1,0,0,0,0,0,.@card;
	} else if (countitem(420231) > 0) {
		delitem 420231,1;
		getitem2 420231,1,1,0,0,0,0,0,.@card;
	}
	return .@lv;
}

function	script	F_md_hol_set_level	{
	.@lv = getarg(0);
	if (.@lv < 1) .@lv = 1;
	if (.@lv > 20) .@lv = 20;
	#MD_HOL_LEVEL = .@lv;
	callfunc("F_md_hol_sync_card");
	return .@lv;
}

function	script	F_md_hol_reward	{
	.@lv = getarg(0);
	setarray .@cycle[1], 15,23,32,42,52,62,72,83,98,118,139,161,183,205,227,250,273,297,321,346;
	setarray .@fruit[1], 8,15,20,26,31,37,43,49,55,62,69,77,86,96,106,118,129,142,158,158;
	setarray .@truth[1], 3,6,21,26,28,31,34,38,42,46,49,53,59,64,70,70,70,70,70,70;
	setarray .@light[1], 5,8,9,11,12,13,14,16,17,18,20,22,26,26,26,26,26,26,26,26;
	// Genesis / Faint Magical Rune from challenge level 9+
	setarray .@genesis[1], 0,0,0,0,0,0,0,0,3,5,7,10,11,12,13,15,16,17,19,22;
	setarray .@rune[1], 0,0,0,0,0,0,0,0,5,6,7,8,10,11,12,13,15,16,17,20;
	if (.@lv < 1) .@lv = 1;
	if (.@lv > 20) .@lv = 20;
	getitem 1001456, .@cycle[.@lv]; // Circulation of Life
	getitem 1001457, .@fruit[.@lv]; // Fruit of Birth
	getitem 1001458, .@fruit[.@lv]; // Fruit of Extinction
	if (.@truth[.@lv] > 0)
		getitem 1001459, .@truth[.@lv]; // Truth of Life
	if (.@light[.@lv] > 0)
		getitem 1001460, .@light[.@lv]; // Light of Life
	if (.@genesis[.@lv] > 0)
		getitem 1001601, .@genesis[.@lv]; // Genesis of Life
	if (.@rune[.@lv] > 0)
		getitem 1001593, .@rune[.@lv]; // Faint Magical Rune Stone
	return;
}

function	script	F_md_hol_boss_hp	{
	.@lv = getarg(0);
	return 800000000 + ((.@lv - 1) * 40000000);
}

function	script	F_md_hol_damage_taken	{
	.@lv = getarg(0);
	.@groggy = getarg(1, 0);
	if (.@groggy)
		return 99; // Near-full damage while stunned after phantom clear (cap 99)
	// Levels 1-3 are easier (kRO)
	switch( .@lv ) {
	case 1: return 10;
	case 2: return 5;
	case 3: return 2;
	}
	return 1; // 99% reduced (DamageTaken percent)
}

function	script	F_md_hol_groggy_sec	{
	.@lv = getarg(0);
	.@sec = 20 - ((.@lv - 1) / 2);
	if (.@sec < 8)
		.@sec = 8;
	return .@sec;
}

function	script	F_md_hol_skill_period	{
	.@lv = getarg(0);
	if (.@lv >= 20)
		return 8;
	if (.@lv >= 10)
		return 10;
	return 12;
}

function	script	F_md_hol_apply_stats	{
	if (!unitexists('hol_boss_gid))
		return;
	.@atk_bonus = ('hol_portal_stack * 800) + (('hol_level - 1) * 500);
	.@def_bonus = ('hol_level - 1) * 8;
	setunitdata 'hol_boss_gid, UMOB_ATKMIN, 24998 + .@atk_bonus;
	setunitdata 'hol_boss_gid, UMOB_ATKMAX, 37102 + .@atk_bonus + ('hol_portal_stack * 200);
	setunitdata 'hol_boss_gid, UMOB_DEF, 358 + .@def_bonus;
	setunitdata 'hol_boss_gid, UMOB_MDEF, 138 + (.@def_bonus / 2);
	.@groggy = ('hol_groggy_until >= gettimetick(2));
	setunitdata 'hol_boss_gid, UMOB_DAMAGETAKEN, callfunc("F_md_hol_damage_taken", 'hol_level, .@groggy);
	if (.@groggy || 'hol_sanctuary)
		setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, ('hol_sanctuary ? 1 : 0);
	else
		setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, 0;
	return;
}

function	script	F_md_hol_heal_rate	{
	// Base heal ticks + death stacks + phantom entrants (kRO snowball)
	.@base = 200 + (('hol_level - 1) * 20);
	.@rate = .@base + ('hol_death_stack * 80) + ('hol_phantom_enter * 40);
	if (.@rate < 50)
		.@rate = 50;
	return .@rate;
}

1@ba_go,1,1,0	script	#hol_boss_ai	-1,{
	end;
OnStart:
	'hol_skill_tick = 0;
	initnpctimer;
	end;
OnStop:
	stopnpctimer;
	end;
OnTimer1000:
	if (('hol_phase != 1 && 'hol_phase != 2) || !unitexists('hol_boss_gid))
		end;
	'hol_skill_tick++;
	getunitdata 'hol_boss_gid, .@b;

	// Phase 1 -> 2 at 50% HP
	if ('hol_phase == 1 && .@b[UMOB_HP] * 100 / .@b[UMOB_MAXHP] <= 50) {
		donpcevent instance_npcname("#hall_of_life_ctrl") + "::OnPhase2";
		initnpctimer;
		end;
	}

	// Groggy window after phantom clear
	if ('hol_groggy_until >= gettimetick(2)) {
		unitstopwalk 'hol_boss_gid, USW_ALL;
		setunitdata 'hol_boss_gid, UMOB_DAMAGETAKEN, 99;
		setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, 0;
		initnpctimer;
		end;
	} else if ('hol_groggy_until > 0 && 'hol_groggy_until < gettimetick(2)) {
		'hol_groggy_until = 0;
		callfunc("F_md_hol_apply_stats");
	}

	// Phase 2 Sanctuary — field under Rigel; immune while standing inside
	if ('hol_phase == 2) {
		.@sanct_cd = 25 - (('hol_level - 1) / 2);
		if (.@sanct_cd < 12)
			.@sanct_cd = 12;
		if ('hol_skill_tick % .@sanct_cd == 0) {
			'hol_sanctuary = 1;
			'hol_sanct_x = .@b[UMOB_X];
			'hol_sanct_y = .@b[UMOB_Y];
			npcskilleffect "AL_SANCTUARY", 10, 'hol_sanct_x, 'hol_sanct_y;
			movenpc instance_npcname("HolSanctuary#field"), 'hol_sanct_x, 'hol_sanct_y;
			enablenpc instance_npcname("HolSanctuary#field");
			donpcevent instance_npcname("HolSanctuary#field") + "::OnPulse";
			setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, 1;
			mapannounce 'hol_map$, "Rigel cast Sanctuary! Lure her out of the field!", bc_map, 0xFF8800;
		}
		if ('hol_sanctuary) {
			if (distance(.@b[UMOB_X], .@b[UMOB_Y], 'hol_sanct_x, 'hol_sanct_y) <= 6) {
				setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, 1;
			} else {
				'hol_sanctuary = 0;
				setunitdata 'hol_boss_gid, UMOB_DMGIMMUNE, 0;
				disablenpc instance_npcname("HolSanctuary#field");
				mapannounce 'hol_map$, "Rigel left Sanctuary — resume the attack!", bc_map, 0x00FF99;
			}
		}
	}

	// Passive HP recovery (snowballs with deaths / phantom entrants)
	if (!'hol_sanctuary && 'hol_groggy_until < gettimetick(2)) {
		.@heal_div = callfunc("F_md_hol_heal_rate");
		.@heal = .@b[UMOB_MAXHP] / .@heal_div;
		if (.@heal > 0 && .@b[UMOB_HP] < .@b[UMOB_MAXHP]) {
			.@newhp = .@b[UMOB_HP] + .@heal;
			if (.@newhp > .@b[UMOB_MAXHP])
				.@newhp = .@b[UMOB_MAXHP];
			setunitdata 'hol_boss_gid, UMOB_HP, .@newhp;
		}
	}

	// Chase nearest player every 2s (unitwalk fails if already walking)
	if ('hol_skill_tick % 2 == 0) {
		getareaunits(BL_PC, 'hol_map$, 0, 0, 400, 100, .@pc[0]);
		.@best = 9999;
		.@tx = .@b[UMOB_X];
		.@ty = .@b[UMOB_Y];
		for ( .@i = 0; .@i < getarraysize(.@pc); ++.@i ) {
			getmapxy .@m$, .@px, .@py, BL_PC, .@pc[.@i];
			if (.@m$ != 'hol_map$)
				continue;
			// Ignore players in phantom space (east room)
			if (.@px >= 300)
				continue;
			.@d = distance(.@px, .@py, .@b[UMOB_X], .@b[UMOB_Y]);
			if (.@d < .@best) {
				.@best = .@d;
				.@tx = .@px;
				.@ty = .@py;
			}
		}
		if (.@best < 9999 && .@best > 2) {
			unitstopwalk 'hol_boss_gid, USW_FORCE_STOP;
			.@dx = (.@tx > .@b[UMOB_X] ? 1 : (.@tx < .@b[UMOB_X] ? -1 : 0));
			.@dy = (.@ty > .@b[UMOB_Y] ? 1 : (.@ty < .@b[UMOB_Y] ? -1 : 0));
			.@wx = .@b[UMOB_X] + (.@dx * 3);
			.@wy = .@b[UMOB_Y] + (.@dy * 3);
			if (.@wx < 36) .@wx = 36;
			if (.@wx > 72) .@wx = 72;
			if (.@wy < 31) .@wy = 31;
			if (.@wy > 67) .@wy = 67;
			unitwalk 'hol_boss_gid, .@wx, .@wy;
		}
	}
	initnpctimer;
	end;
}
