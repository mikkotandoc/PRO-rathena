function	script	F_Aircrash_HpMult	{
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return 1;
	getpartymember .@party_id;
	return max(1, min($@partymembercount, 10));
}

function	script	F_Aircrash_SpawnMob	{
	.@map$ = getarg(0);
	.@label$ = getarg(1);
	.@mult = getarg(2);
	setarray .@ids[0], 20886, 20887, 20888, 20889, 20890;
	.@id = .@ids[ rand(getarraysize(.@ids)) ];
	// Spawn in walkable cavity zones (areamonster picks valid cells).
	switch (rand(6)) {
	case 0: areamonster .@map$,  30,  30,  70,  70, "--ja--", .@id, 1, .@label$; break;
	case 1: areamonster .@map$, 230,  30, 270,  70, "--ja--", .@id, 1, .@label$; break;
	case 2: areamonster .@map$,  30, 230,  70, 270, "--ja--", .@id, 1, .@label$; break;
	case 3: areamonster .@map$, 230, 230, 270, 270, "--ja--", .@id, 1, .@label$; break;
	case 4: areamonster .@map$, 120,  20, 180,  60, "--ja--", .@id, 1, .@label$; break;
	default: areamonster .@map$, 120, 240, 180, 280, "--ja--", .@id, 1, .@label$; break;
	}
	if (.@mult > 1 && $@mobid[0]) {
		.@hp = getunitdata($@mobid[0], UMOB_MAXHP);
		setunitdata $@mobid[0], UMOB_MAXHP, .@hp * .@mult;
		setunitdata $@mobid[0], UMOB_HP, .@hp * .@mult;
	}
	return;
}

function	script	F_Aircrash_ResetHunt	{
	#aircrash_hunt = 0;
	#aircrash_kills = 0;
	return;
}

		next;
		mes "[Professor]";
		mes "When you are ready, register your party to begin the investigation.";
		mes "^ff0000Warning: monsters will begin spawning once you register.^000000";
		next;
		if (select("Listen only.:Register investigation party.") == 1) {
			mes "[Professor]";
			mes "Take your time. Speak with my students for more details.";
			close;
		}
		if (getcharid(1) == 0 || is_party_leader() == false) {
			mes "[Professor]";
			mes "Only the party leader may register the investigation.";
			close;
		}
		'investigation = 1;
		'hp_mult = callfunc("F_Aircrash_HpMult");
		donpcevent instance_npcname("#AircrashCtrl") + "::OnStartSpawns";
		mapannounce instance_mapname("1@mjo1"), "Professor: Investigation registered. Contaminated monsters will now appear!", bc_map, 0xFFFF00;
		mes "[Professor]";
		mes "Registration complete. Be careful out there.";
		close;
	}
	if (#aircrash_hunt > 0) {
		.@need = 0;
		.@reward = 0;
		switch(#aircrash_hunt) {
		case 1: .@need = 100; .@reward = 1; break;
		case 2: .@need = 200; .@reward = 2; break;
		case 3: .@need = 350; .@reward = 3; break;
		}
		if (#aircrash_kills < .@need) {
			mes "Progress: ^0000cd" + #aircrash_kills + " / " + .@need + "^000000 monsters eliminated.";
			close;
		}
		getitem 1000363, .@reward;
		if (#aircrash_hunt_tier < #aircrash_hunt)
			#aircrash_hunt_tier = #aircrash_hunt;
		callfunc "F_Aircrash_ResetHunt";
		mes "[Professor]";
		mes "Excellent work. Here are your Ymir Fragments.";
		close;
	}
	mes "The investigation is underway.";
	mes "Accept hunting orders from the student, then return here to claim Ymir Fragments.";
	close;
}

1@mjo1,1,1,0	script	#AircrashCtrl	-1,{
	end;

OnInstanceInit:
	'investigation = 0;
	'boss_summoned = 0;
	'hp_mult = 1;
	'map2$ = instance_mapname("1@mjo2");
	end;

OnStartSpawns:
	'map2$ = instance_mapname("1@mjo2");
	.@label$ = instance_npcname("#AircrashCtrl") + "::OnMobDead";
	for (.@i = 0; .@i < 40; .@i++)
		callfunc("F_Aircrash_SpawnMob", 'map2$, .@label$, 'hp_mult);
	mapannounce 'map2$, "Contaminated monsters have appeared in the cavity!", bc_map, 0xFFFF00;
	end;

OnSummonBoss:
	'map2$ = instance_mapname("1@mjo2");
	if (mobcount('map2$, instance_npcname("#AircrashCtrl") + "::OnBossDead") > 0)
		end;
	.@label$ = instance_npcname("#AircrashCtrl") + "::OnBossDead";
	monster 'map2$, 150, 150, "--ja--", 20891, 1, .@label$;
	if ('hp_mult > 1 && $@mobid[0]) {
		.@hp = getunitdata($@mobid[0], UMOB_MAXHP);
		setunitdata $@mobid[0], UMOB_MAXHP, .@hp * 'hp_mult;
		setunitdata $@mobid[0], UMOB_HP, .@hp * 'hp_mult;
	}
	mapannounce 'map2$, "An Unidentified Creature has appeared!", bc_map, 0xFF0000;
	end;

OnMobDead:
	if ('investigation == 0)
		end;
	if (playerattached()) {
		if (#aircrash_hunt > 0 && #aircrash_kills < 350)
			#aircrash_kills++;
	}
	'map2$ = instance_mapname("1@mjo2");
	if (mobcount('map2$, instance_npcname("#AircrashCtrl") + "::OnMobDead") < 25)
		callfunc("F_Aircrash_SpawnMob", 'map2$, instance_npcname("#AircrashCtrl") + "::OnMobDead", 'hp_mult);
	end;

OnBossDead:
	mapannounce 'map2$, "The Unidentified Creature has been defeated!", bc_map, 0x00FF99;
	end;
}
