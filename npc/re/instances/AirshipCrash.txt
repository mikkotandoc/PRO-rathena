function	script	F_Aircrash_HpMult	{
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return 1;
	getpartymember .@party_id;
	return max(1, min($@partymembercount, 10));
}

function	script	F_Aircrash_SpawnMob	{
	.@map$ = getarg(0);
	.@label$ = getarg(1);
	.@mult = getarg(2);
	setarray .@ids[0], 20886, 20887, 20888, 20889, 20890;
	.@id = .@ids[ rand(getarraysize(.@ids)) ];
	// Spawn in walkable cavity zones (areamonster picks valid cells).
	switch (rand(6)) {
	case 0: areamonster .@map$,  30,  30,  70,  70, "--ja--", .@id, 1, .@label$; break;
	case 1: areamonster .@map$, 230,  30, 270,  70, "--ja--", .@id, 1, .@label$; break;
	case 2: areamonster .@map$,  30, 230,  70, 270, "--ja--", .@id, 1, .@label$; break;
	case 3: areamonster .@map$, 230, 230, 270, 270, "--ja--", .@id, 1, .@label$; break;
	case 4: areamonster .@map$, 120,  20, 180,  60, "--ja--", .@id, 1, .@label$; break;
	default: areamonster .@map$, 120, 240, 180, 280, "--ja--", .@id, 1, .@label$; break;
	}
	if (.@mult > 1 && $@mobid[0]) {
		.@hp = getunitdata($@mobid[0], UMOB_MAXHP);
		setunitdata $@mobid[0], UMOB_MAXHP, .@hp * .@mult;
		setunitdata $@mobid[0], UMOB_HP, .@hp * .@mult;
	}
	return;
}

function	script	F_Aircrash_ResetHunt	{
	#aircrash_hunt = 0;
	#aircrash_kills = 0;
	return;
}

function	script	F_Aircrash_HelpInjured	{
	if ('injured_rescued >= 5) {
		mes "^0000cdYou have already helped enough survivors to weaken the creature.^000000";
		close;
	}
	mes "^0000cdAn injured passenger lies among the debris.^000000";
	next;
	if (select("Help the injured person.:Leave them for now.") == 2)
		close;
	specialeffect EF_RESURRECTION, AREA, strnpcinfo(0);
	sleep2 500;
	'injured_rescued++;
	mapannounce instance_mapname("1@mjo1"), "Injured Passenger: Thank you... I can walk on my own now...", bc_map, 0xFFFF00;
	disablenpc strnpcinfo(0);
	if ('injured_rescued >= 5)
		mapannounce instance_mapname("1@mjo1"), "The survivors you helped may weaken the contaminated creature lurking in the cave.", bc_map, 0x00FF99;
	close;
}

1@mjo1,1,1,0	script	#AircrashCtrl	-1,{
	end;

OnInstanceInit:
	'investigation = 0;
	'boss_summoned = 0;
	'hp_mult = 1;
	'injured_rescued = 0;
	'pelloch_saved = 0;
	'map2$ = instance_mapname("1@mjo2");
	end;

OnStartSpawns:
	'map2$ = instance_mapname("1@mjo2");
	.@label$ = instance_npcname("#AircrashCtrl") + "::OnMobDead";
	for (.@i = 0; .@i < 40; .@i++)
		callfunc("F_Aircrash_SpawnMob", 'map2$, .@label$, 'hp_mult);
	mapannounce 'map2$, "Contaminated monsters have appeared in the cavity!", bc_map, 0xFFFF00;
	end;

OnSummonBoss:
	'map2$ = instance_mapname("1@mjo2");
	if (mobcount('map2$, instance_npcname("#AircrashCtrl") + "::OnBossDead") > 0)
		end;
	.@label$ = instance_npcname("#AircrashCtrl") + "::OnBossDead";
	// kRO: helping 5 injured survivors summons the weaker boss (20891).
	.@boss_id = ('injured_rescued >= 5 ? 20891 : 21062);
	monster 'map2$, 150, 150, "--ja--", .@boss_id, 1, .@label$;
	if ('hp_mult > 1 && $@mobid[0]) {
		.@hp = getunitdata($@mobid[0], UMOB_MAXHP);
		setunitdata $@mobid[0], UMOB_MAXHP, .@hp * 'hp_mult;
		setunitdata $@mobid[0], UMOB_HP, .@hp * 'hp_mult;
	}
	if (.@boss_id == 20891)
		mapannounce 'map2$, "A weakened Unidentified Creature has appeared!", bc_map, 0xFF0000;
	else
		mapannounce 'map2$, "A powerful Unidentified Life Form has appeared!", bc_map, 0xFF0000;
	end;

OnMobDead:
	if ('investigation == 0)
		end;
	if (playerattached()) {
		if (#aircrash_hunt > 0 && #aircrash_kills < 350)
			#aircrash_kills++;
	}
	'map2$ = instance_mapname("1@mjo2");
	if (mobcount('map2$, instance_npcname("#AircrashCtrl") + "::OnMobDead") < 25)
		callfunc("F_Aircrash_SpawnMob", 'map2$, instance_npcname("#AircrashCtrl") + "::OnMobDead", 'hp_mult);
	end;

OnBossDead:
	mapannounce 'map2$, "The Unidentified Creature has been defeated!", bc_map, 0x00FF99;
	end;
}
