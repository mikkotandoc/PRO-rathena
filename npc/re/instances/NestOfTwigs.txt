function	script	F_ep20_twig_clear	{
	if (ep20_main == 25 && isbegin_quest(23106) == 1) {
		callfunc("F_ep20_completequest", 23106);
		ep20_main = 26;
		callfunc("F_ep20_setquest", 23109);
		callfunc("F_ep20_reward", 10, 70);
	}
	return;
}

jor_twice,1,1,0	script(CLOAKED)	Nest of Twigs#ep20	HIDDEN_WARP_NPC,{
	end;

OnStart:
	if (ep20_main < 25) end;
	callfunc("F_ep20_instance_enter", "Nest of Twigs");
	end;
}

1@twig,100,100,3	script	Lehar#ep20twig01	4_EP19_LEHAR,{
	if (ep20_main != 25) end;
	cutin "ep19_lehar01.png",2;
	mes "[Lehar]";
	mes "Guardians ahead! Stay close!";
	close2;
	cutin "",255;
	.@label$ = instance_npcname("Lehar#ep20twig01") + "::OnGuardDead";
	monster "1@twig",105,100,"Guardian",21990,3,.@label$;
	end;

OnGuardDead:
	if (mobcount("1@twig", instance_npcname("Lehar#ep20twig01") + "::OnGuardDead") > 0)
		end;
	cutin "ep19_leon01.png",0;
	mes "[Leon]";
	mes "The twig nest entrance is collapsing. We did it!";
	callfunc("F_ep20_twig_clear");
	close2;
	cutin "",255;
	instance_destroy;
	warp "jor_safty1",328,150;
	end;
}
