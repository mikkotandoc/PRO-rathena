//===== rAthena Script =======================================
//= Bio 5: Bed of Honor (Hard Mode)
//===== Description: =========================================
//= Memorial instance for high-end characters (BL 250+).
//= MuhRO/kRO Bed of Honor mechanics: 110 kills -> true form,
//= 4 true kills of the same type -> MVP. Rewards per MuhRO.
//============================================================

function	script	F_Bio5Hard_Instance	{
	.@name$ = "Bio 5 Bed of Honor Hard";
	if (BaseLevel < 250) {
		mes "[Ohno Tohiro]";
		mes "The amplified tomb simulation is only for adventurers with base level 250 or higher.";
		close;
	}
	if (checkquest(90520, PLAYTIME) == 2)
		erasequest 90520;
	if (checkquest(90522, PLAYTIME) == 2)
		erasequest 90522;

	mes "[Ohno Tohiro]";
	mes "Beyond the public tomb lies a ^0000cdsealed memory echo^000000 Rekenber studied in secret.";
	mes "Stronger fallen champions respawn inside your own copy of the Bed of Honor.";
	next;
	mes "[Ohno Tohiro]";
	mes "Entry costs ^0000cd200,000 Zeny^000000 and ^0000cd20 Sentimental Fragments^000000.";
	mes "You may enter once every ^0000cd4 hours^000000.";
	next;
	switch (select("Create Hard Mode instance", "Enter Hard Mode instance", "Claim Hard Mode hunt reward", "Cancel")) {
	case 1:
		if (!is_party_leader()) {
			mes "[Ohno Tohiro]";
			mes "Only the party leader can open the sealed echo.";
			close;
		}
		switch (checkquest(90520, PLAYTIME)) {
		case 0:
		case 1:
			mes "[Ohno Tohiro]";
			mes "You are still recovering from the last hard simulation.";
			close;
		}
		switch (instance_create(.@name$)) {
		case -3:
			dispbottom "Memorial Dungeon '" + .@name$ + "' is already in progress.", 0xFFFFFF;
			close;
		case -4:
		case -2:
		case -1:
			mes "[Ohno Tohiro]";
			mes "^0000FF" + .@name$ + "^000000 - instance creation failed.";
			close;
		}
		mes "[Ohno Tohiro]";
		mes "The echo is ready. Gather your party and enter when prepared.";
		close;
	case 2:
		switch (checkquest(90520, PLAYTIME)) {
		case 0:
		case 1:
			mes "[Ohno Tohiro]";
			mes "You are still on cooldown from the last hard simulation.";
			close;
		}
		if (Zeny < 200000 || countitem(22687) < 20) {
			mes "[Ohno Tohiro]";
			mes "You need 200,000 Zeny and 20 Sentimental Fragments to enter the echo.";
			close;
		}
		switch (instance_enter(.@name$)) {
		case IE_OTHER:
			mes "[Ohno Tohiro]";
			mes "An unknown error occurred.";
			close;
		case IE_NOINSTANCE:
			mes "[Ohno Tohiro]";
			mes "There is no open hard mode instance. Create one first.";
			close;
		case IE_OK:
			Zeny -= 200000;
			delitem 22687, 20;
			setquest 90520;
			if (checkquest(90521, HUNTING) != -1)
				erasequest 90521;
			setquest 90521;
			mapannounce "lighthalzen", strcharinfo(0) + " is entering " + .@name$ + ".", bc_map, "0xFF8800", FW_NORMAL, 12;
			end;
		}
	case 3:
		callsub S_Reward;
	case 4:
		close;
	}
	end;

S_Reward:
	switch (checkquest(90522, PLAYTIME)) {
	case 0:
	case 1:
		mes "[Ohno Tohiro]";
		mes "You already claimed today's hard mode hunt stipend.";
		close;
	case 2:
		erasequest 90522;
		break;
	}
	switch (checkquest(90521, HUNTING)) {
	case -1:
		mes "[Ohno Tohiro]";
		mes "Take the hard mode hunt first by entering the memorial echo.";
		close;
	case 0:
	case 1:
		mes "[Ohno Tohiro]";
		mes "Hunt ^0000cd200^000000 monsters inside the hard Bed of Honor, then return.";
		close;
	case 2:
		if (!checkweight(6820, 5) || !checkweight(23016, 3) || !checkweight(25131, 2)) {
			mes "[Ohno Tohiro]";
			mes "Your inventory is too full to receive the reward.";
			close;
		}
		erasequest 90521;
		setquest 90522;
		getitem 6820, 5;      // Energy Fragment
		getitem 23016, 3;     // Grudge Fragment
		getitem 25131, 2;     // Fallen Energy Particle
		getexp 500000000, 300000000;
		mes "[Ohno Tohiro]";
		mes "Excellent work in the amplified tomb. Here is your research stipend.";
		close;
	}
	end;
}

lighthalzen,323,218,4	script	Bed of Honor Hard#lhz	4_M_RASWORD,{
	mes "[Ohno Tohiro]";
	mes "If you are strong enough, I can open Rekenber's sealed hard simulation of the Bed of Honor.";
	next;
	callfunc "F_Bio5Hard_Instance";
	end;

OnInit:
	setunittitle getnpcid(0), "<Bed of Honor Hard>";
	end;
}

lhz_dun_n,1,1,0	script	#bio5h_ctrl	HIDDEN_WARP_NPC,{
	end;

OnInstanceInit:
	'map$ = instance_mapname("lhz_dun_n");
	'boss_active = 0;
	deletearray 'regular_kill;
	deletearray 'true_kill;
	disablenpc instance_npcname("lhz_dun_n");
	killmonster 'map$, "";
	callsub S_SpawnRegular;
	end;

S_SpawnRegular:
	// MuhRO tomb roster — regular forms only; true/MVP are spawned by kill counters.
	monster 'map$,0,0,"--en--",3208,20, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3208";
	monster 'map$,0,0,"--en--",3209,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3209";
	monster 'map$,0,0,"--en--",3210,15, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3210";
	monster 'map$,0,0,"--en--",3211,20, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3211";
	monster 'map$,0,0,"--en--",3212,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3212";
	monster 'map$,0,0,"--en--",3213,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3213";
	monster 'map$,0,0,"--en--",3226,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3226";
	monster 'map$,0,0,"--en--",3227,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3227";
	monster 'map$,0,0,"--en--",3228,15, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3228";
	monster 'map$,0,0,"--en--",3229,15, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3229";
	monster 'map$,0,0,"--en--",3230,15, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3230";
	monster 'map$,0,0,"--en--",3231,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3231";
	monster 'map$,0,0,"--en--",3232,10, instance_npcname("#bio5h_ctrl") + "::OnRegularDead3232";
	end;

OnRegularDead3208: callsub S_Regular, 3208;
OnRegularDead3209: callsub S_Regular, 3209;
OnRegularDead3210: callsub S_Regular, 3210;
OnRegularDead3211: callsub S_Regular, 3211;
OnRegularDead3212: callsub S_Regular, 3212;
OnRegularDead3213: callsub S_Regular, 3213;
OnRegularDead3226: callsub S_Regular, 3226;
OnRegularDead3227: callsub S_Regular, 3227;
OnRegularDead3228: callsub S_Regular, 3228;
OnRegularDead3229: callsub S_Regular, 3229;
OnRegularDead3230: callsub S_Regular, 3230;
OnRegularDead3231: callsub S_Regular, 3231;
OnRegularDead3232: callsub S_Regular, 3232;

S_Regular:
	.@regular = getarg(0);
	if ('boss_active || 'regular_kill[.@regular] >= 110)
		end;
	'regular_kill[.@regular]++;
	if ('regular_kill[.@regular] < 110)
		end;
	.@true = .@regular + 6 + (.@regular > 3219);
	mapannounce 'map$, "Unknown Voice: Feel my true power.", bc_map, 0xFF5555;
	monster 'map$,0,0,"--en--", .@true, 1, instance_npcname("#bio5h_ctrl") + "::OnTrueDead" + .@true;
	end;

OnTrueDead3214: callsub S_True, 3214;
OnTrueDead3215: callsub S_True, 3215;
OnTrueDead3216: callsub S_True, 3216;
OnTrueDead3217: callsub S_True, 3217;
OnTrueDead3218: callsub S_True, 3218;
OnTrueDead3219: callsub S_True, 3219;
OnTrueDead3233: callsub S_True, 3233;
OnTrueDead3234: callsub S_True, 3234;
OnTrueDead3235: callsub S_True, 3235;
OnTrueDead3236: callsub S_True, 3236;
OnTrueDead3237: callsub S_True, 3237;
OnTrueDead3238: callsub S_True, 3238;
OnTrueDead3239: callsub S_True, 3239;

S_True:
	.@true = getarg(0);
	if ('boss_active)
		end;
	'true_kill[.@true]++;
	.@regular = .@true - 6 - (.@true > 3219);
	'regular_kill[.@regular] = 0;
	mapannounce 'map$, "Unknown Voice: You murdered my minion!", bc_map, 0xFF5555;
	if ('true_kill[.@true] < 4)
		end;
	.@mvp = .@true + 6 + (.@true > 3239);
	mapannounce 'map$, "Unknown Voice: Feel my vengeance!", bc_map, 0xFF0000;
	monster 'map$,140,230,"--en--", .@mvp, 1, instance_npcname("#bio5h_ctrl") + "::OnMvpDead";
	'boss_active = 1;
	end;

OnMvpDead:
	mapannounce 'map$, "Unknown Voice: See you soon.", bc_map, 0x00FF00;
	'boss_active = 0;
	deletearray 'regular_kill;
	deletearray 'true_kill;
	.@count = getmapunits(BL_PC, 'map$, .@aid);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!attachrid(.@aid[.@i]))
			continue;
		getitem 25131, 3;
		getitem 23016, 1;
		getexp 150000000, 90000000;
		.@roll = rand(10000);
		if (.@roll < 50)
			getitem 400078, 1;
		else if (.@roll < 100)
			getitem 400116, 1;
		else if (.@roll < 125)
			getitem 20748, 1;
		else if (.@roll < 175)
			getitem 100753, 1;
		else if (.@roll < 225)
			getitem 100821, 1;
		detachrid;
	}
	initnpctimer;
	end;

OnTimer60000:
	stopnpctimer;
	callsub S_SpawnRegular;
	end;
}
