function	script	F_ep20_checkTransform	{
	if (getstatus(SC_MONSTER_TRANSFORM,1) != 21530) {
		mes "[" + getarg(0) + "]";
		mes "Human?! Go away.";
		close2;
		warp "jor_back3",65,321;
		end;
	}
	return;
}

function	script	F_ep20_reward	{
	.@whiskers = getarg(0);
	.@rep = getarg(1);
	if (.@whiskers > 0)
		getitem 1001217, .@whiskers;	// Ep20_Cat_Whiskers
	if (.@rep > 0)
		add_reputation_points( REPUTATION_EP20, .@rep );
	return;
}

function	script	F_ep20_setquest	{
	if (isbegin_quest(getarg(0)) == 0)
		setquest getarg(0);
	return;
}

// Returns true when the player has a solo party with at least one online member.
function	script	F_ep20_instance_party_ok	{
	.@party_id = getcharid(1);

	if (!.@party_id)
		return 0;
	if (!instance_check_party(.@party_id, 1))
		return 0;
	getpartymember .@party_id, 2;
	if ($@partymembercount != 1)
		return 0;
	return 1;
}

// getarg(0) = instance name, getarg(1) = 0 auto prepare+enter, 1 prepare only, 2 enter only
function	script	F_ep20_instance_enter	{
	.@name$ = getarg(0);
	.@mode = getarg(1, 0);

	if (!callfunc("F_ep20_instance_party_ok")) {
		mes "- ^0000FFCreate a one-person party with ^4D4DFF/organize PARTY NAME^0000FF, then try again.^000000 -";
		close3;
	}

	if (.@mode == 1 || .@mode == 0) {
		.@create = instance_create(.@name$);
		if (.@create == -3)
			instance_warning(2, .@name$);
		else if (.@create < 0) {
			switch (.@create) {
			case -1:
				mes "This memorial dungeon is not registered on the server.";
				mes "Ask a GM to reload the instance database.";
				break;
			case -2:
				mes "Your party could not be found. Reform your solo party and try again.";
				break;
			case -4:
				mes "The server has no free instance slots right now. Try again shortly.";
				break;
			default:
				mes "Could not create the memorial dungeon. (Error: " + .@create + ")";
				break;
			}
			close3;
		}
		if (.@mode == 1) {
			mes "- ^0000FFEntry preparations have begun. Select Enter when ready.^000000 -";
			close3;
		}
	}

	if (instance_enter(.@name$) == IE_OK)
		end;
	if (.@mode == 2) {
		mes "- ^0000FFThe memorial dungeon has not been created yet, or it is still loading. Prepare it first, wait a moment, then enter again.^000000 -";
		close3;
	}
	mes "- ^0000FFThe memorial dungeon is still loading. Wait a moment and interact again.^000000 -";
	close3;
}

function	script	F_ep20_startHoruru	{
	if (checkweight(1001217,20) == 0) {
		mes "You cannot proceed with the conversation due to the large number of items you have.";
		mes "Please clean up your items and try again.";
		close;
	}
	if (isbegin_quest(17690) >= 1 || isbegin_quest(16691) >= 1) {
		if (ep20_main == 0)
			ep20_main = 1;
		callfunc("F_ep20_horuruProgress");
		end;
	}
	if (!callfunc("F_ep20_canStart")) {
		cutin "ep19_iwin06.png",2;
		mes "[Horuru]";
		if (ep19_main < 100) {
			mes "We still have unfinished business regarding the Rgans' hideout.";
			mes "Please finish the story in Issgard first.";
		} else if (BaseLevel < 215) {
			mes "The Ancient Ice Gorge is too harsh for you right now.";
			mes "Come back when you reach Base Level 215.";
		} else if (isbegin_quest(17650) == 1) {
			mes "You accepted the surveillance request but haven't finished the briefing outside yet.";
			mes "Meet me at ^0000CDIce Castle (27, 126)^000000 to continue.";
		} else {
			mes "Please help us with the surveillance around the snake's nest first.";
			mes "Talk to me at the ^0000CDrestaurant in Ice Castle^000000 to begin.";
		}
		close3;
	}
	cutin "ep19_iwin06.png",2;
	mes "[Horuru]";
	mes "Adventurer, the Rgans sealed the Warmth of the Snake and vanished.";
	mes "We can't find Lasgand or Bagot anywhere on the ice plains.";
	next;
	mes "[Horuru]";
	mes "We've decided to enter the ^0000CDAncient Ice Gorge^000000, a place that was forbidden until now.";
	mes "Will you come with us?";
	next;
	if (select( "I'll help.", "Not now." ) == 2) {
		mes "[Horuru]";
		mes "I understand. Come back when you're ready.";
		close3;
	}
	mes "[Horuru]";
	mes "Thank you. Leon is gathering everyone in the castle.";
	mes "Meet us at the conference room.";
	callfunc("F_ep20_setquest", 20000);
	callfunc("F_ep20_setquest", 17690);
	ep20_main = 1;
	callfunc("F_ep20_reward", 20, 90);
	close3;
}

	close3;
}
