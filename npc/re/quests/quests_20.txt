function	script	F_ep20_checkTransform	{
	if (getstatus(SC_MONSTER_TRANSFORM,1) != 21530) {
		mes "[" + getarg(0) + "]";
		mes "Human?! Go away.";
		close2;
		warp "jor_back3",65,321;
		end;
	}
	return;
}

function	script	F_ep20_reward	{
	.@whiskers = getarg(0);
	.@rep = getarg(1);
	if (.@whiskers > 0)
		getitem 1001217, .@whiskers;	// Ep20_Cat_Whiskers
	if (.@rep > 0)
		add_reputation_points( REPUTATION_EP20, .@rep );
	return;
}

function	script	F_ep20_setquest	{
	if (isbegin_quest(getarg(0)) == 0)
		setquest getarg(0);
	return;
}

function	script	F_ep20_completequest	{
	if (isbegin_quest(getarg(0)) == 1)
		completequest getarg(0);
	return;
}

function	script	F_ep20_safetyMenu	{
	mes "[Safety Zone Portal]";
	mes "Select a destination.";
	next;
	switch( select( "Safety Zone 1 (64, 328)", "Safety Zone 2 (195, 327)", "Safety Zone 3 (330, 327)", "Southern Camp (195, 155)", "Deep Camp (328, 150)", "Ancient Ice Gorge East", "Cancel" ) ) {
	case 1:
		warp "jor_safty1", 64, 328;
		end;
	case 2:
		warp "jor_safty1", 195, 327;
		end;
	case 3:
		warp "jor_safty1", 330, 327;
		end;
	case 4:
		warp "jor_safty1", 195, 155;
		end;
	case 5:
		warp "jor_safty1", 328, 150;
		end;
	case 6:
		warp "jor_back5", 350, 75;
		end;
	case 7:
		close;
	}
	end;
}

function	script	F_ep20_canStart	{
	return (ep19_main == 100 && BaseLevel >= 215 && isbegin_quest(17650) == 2);
}

function	script	F_ep20_startHoruru	{
	if (checkweight(1001217,20) == 0) {
		mes "You cannot proceed with the conversation due to the large number of items you have.";
		mes "Please clean up your items and try again.";
		close;
	}
	if (isbegin_quest(17700) >= 1) {
		if (ep20_main == 0)
			ep20_main = 1;
		callfunc("F_ep20_horuruProgress");
		end;
	}
	if (!callfunc("F_ep20_canStart")) {
		cutin "ep19_iwin06.png",2;
		mes "[Horuru]";
		if (ep19_main < 100) {
			mes "We still have unfinished business regarding the Rgans' hideout.";
			mes "Please finish the story in Issgard first.";
		} else if (BaseLevel < 215) {
			mes "The Ancient Ice Gorge is too harsh for you right now.";
			mes "Come back when you reach Base Level 215.";
		} else if (isbegin_quest(17650) == 1) {
			mes "You accepted the surveillance request but haven't finished the briefing outside yet.";
			mes "Meet me at ^0000CDIce Castle (27, 126)^000000 to continue.";
		} else {
			mes "Please help us with the surveillance around the snake's nest first.";
			mes "Talk to me at the ^0000CDrestaurant in Ice Castle^000000 to begin.";
		}
		close3;
	}
	cutin "ep19_iwin06.png",2;
	mes "[Horuru]";
	mes "Adventurer, the Rgans sealed the Warmth of the Snake and vanished.";
	mes "We can't find Lasgand or Bagot anywhere on the ice plains.";
	next;
	mes "[Horuru]";
	mes "We've decided to enter the ^0000CDAncient Ice Gorge^000000, a place that was forbidden until now.";
	mes "Will you come with us?";
	next;
	if (select( "I'll help.", "Not now." ) == 2) {
		mes "[Horuru]";
		mes "I understand. Come back when you're ready.";
		close3;
	}
	mes "[Horuru]";
	mes "Thank you. Leon is gathering everyone in the castle.";
	mes "Meet us at the conference room.";
	callfunc("F_ep20_setquest", 20000);
	callfunc("F_ep20_setquest", 17700);
	ep20_main = 1;
	callfunc("F_ep20_reward", 20, 90);
	close3;
}

function	script	F_ep20_horuruProgress	{
	cutin "ep19_iwin06.png",2;
	mes "[Horuru]";
	if (ep20_main == 1)
		mes "Let's go inside the castle. Leon is waiting for us at the conference room.";
	else
		mes "The Ancient Ice Gorge is a dangerous place. Be careful.";
	close3;
}
