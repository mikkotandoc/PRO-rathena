function	script	F_ep20_sanctCheck	{
	callfunc("F_ep20_checkTransform", getarg(0));
	return;
}

// Minimap Rgan trust marks after Pamoshgand (official steps 34-43). ep20_sanct_mark 0-9.
function	script	F_ep20_sanctMarkHelp	{
	.@step = getarg(0);
	.@mes$ = getarg(1);
	.@next_map$ = getarg(2);
	.@next_x = getarg(3);
	.@next_y = getarg(4);

	callfunc("F_ep20_sanctCheck","Rgan");
	if (ep20_main != 19 || isbegin_quest(17701) != 2)
		end;
	if (ep20_sanct_mark > .@step) {
		mes "[Rgan]";
		mes "You already helped here.";
		close;
	}
	if (ep20_sanct_mark < .@step) {
		mes "[Rgan]";
		mes "Another marked Rgan needs you first. Follow the quest marks on your minimap.";
		close;
	}
	mes "[Rgan]";
	mes .@mes$;
	ep20_sanct_mark = .@step + 1;
	if (.@next_map$ != "")
		navigateto(.@next_map$, .@next_x, .@next_y);
	else {
		mes "Report to ^0000CDSerekgand (157, 171)^000000.";
		navigateto("jor_sanct", 157, 171);
	}
	close;
}

function	script	F_ep20_snakeStrawberryGather	{
	if (ep20_main != 19 || isbegin_quest(17698) != 1)
		end;
	.@need = 3 - countitem(1001154);	// Ep20_Jor_Berry (Snake Strawberry)
	if (.@need > 0) {
		mes "You gather Snake Strawberries from the firm spot.";
		getitem 1001154, .@need;
	}
	if (countitem(1001154) < 3) {
		mes "You need ^0000CD3 Snake Strawberries^000000. Check the other firm spots on the minimap.";
		close;
	}
	mes "You collected ^0000CD3 Snake Strawberries^000000.";
	mes "Report to ^0000CDLazy (155, 85)^000000 in Sacred Roots.";
	navigateto("jor_sanct", 155, 85);
	close;
}

// Main quest step 35 - Lazy assigns Snake Strawberry collection (quest 17697 -> 17698)
jor_sanct,155,85,5	script	Lazy#ep20sanct01	4_EP19_LAZY,{
	callfunc("F_ep20_checkTransform","Lazy");
	if (ep20_main < 19) end;
	if (ep20_main > 19) {
		mes "[Lazy]";
		mes "Keep earning the bishops' trust. Serekgand is near the maze door.";
		close;
	}
	if (isbegin_quest(17697) == 1) {
		mes "[Lazy]";
		mes "We need Snake Strawberries from the firm spots in ^0000CDSnake God's Root 2F^000000.";
		mes "Collect ^0000CD3 Snake Strawberries^000000 and bring them back to me.";
		callfunc("F_ep20_completequest", 17697);
		callfunc("F_ep20_setquest", 17698);
		navigateto("jor_root2", 234, 248);
		close;
	}
	if (isbegin_quest(17698) == 1) {
		if (countitem(1001154) < 3) {
			mes "[Lazy]";
			mes "I still need ^0000CD3 Snake Strawberries^000000 from the firm spots in Root 2F.";
			navigateto("jor_root2", 234, 248);
			close;
		}
		delitem 1001154, 3;
		mes "[Lazy]";
		mes "Perfect. Now help the bishops marked on your minimap to earn their trust.";
		callfunc("F_ep20_completequest", 17698);
		callfunc("F_ep20_setquest", 17699);
		close;
	}
	mes "[Lazy]";
	mes "Help the bishops and faithful Rgans marked on your minimap.";
	mes "When everyone trusts you, see ^0000CDSerekgand (157, 171)^000000.";
	close;
OnInit:
	questinfo( QTYPE_QUEST, QMARK_YELLOW, "ep20_main == 19 && isbegin_quest(17697) == 1" );
	questinfo( QTYPE_QUEST2, QMARK_YELLOW, "ep20_main == 19 && isbegin_quest(17698) == 1 && countitem(1001154) >= 3" );
	end;
}
