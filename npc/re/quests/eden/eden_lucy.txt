//===== rAthena Script =======================================
//= Paradise Shadow Equipment Rental: Lucy
//===== Description: =========================================
//= kRO/iRO-aligned shadow gear rental at moc_para01 110/91.
//= Advanced class weapons use eaclass() + EAJ_THIRDMASK (official).
//============================================================

moc_para01,110,91,4	script	Lucy	70,{
	mes "[Lucy]";
	mes "Greetings, adventurer! I handle the distribution of Paradise Shadow items.";
	mes "Looking for simple training rentals, or advanced combat pieces?";
	next;

	if (!callfunc("F_HasEdenGroupMark")) {
		mes "[Lucy]";
		mes "Access denied. Only active members of the Eden Group can rent shadow items.";
		close;
	}

	// kRO menu order: Advanced (1), Beginner (2), Cancel (3)
	switch(select("Rent Advanced Shadow Gear (Lv 100+):Rent Beginner Shadow Gear (Lv 10 - Free):Cancel")) {

	case 1:
		mes "[Lucy]";
		if (BaseLevel < 100 || !callfunc("F_IsRenewalJob")) {
			mes "Access Denied.";
			mes "Advanced Shadow Equipment rentals strictly require **Base Level 100** and a **Third Job or Advanced Extended Class**.";
			close;
		}

		mes "Advanced rental pieces cost **20 Eden Coins** and last for **72 hours (3 days)**.";
		next;

		switch(select("Rent Advanced Shadow Armor/Accessory:Rent Class Shadow Weapon")) {
		case 1:
			mes "[Lucy]";
			mes "Select your defensive piece:";
			next;
			switch(select("Paradise Advanced Shadow Shield:Paradise Advanced Shadow Armor:Paradise Advanced Shadow Shoes:Paradise Advanced Shadow Earring:Paradise Advanced Shadow Pendant")) {
				case 1: .@shadow_id = 24693; break; // S_2Para_Shield
				case 2: .@shadow_id = 24694; break; // S_2Para_Armor
				case 3: .@shadow_id = 24695; break; // S_2Para_Shoes
				case 4: .@shadow_id = 24696; break; // S_2Para_Earing
				case 5: .@shadow_id = 24697; break; // S_2Para_Pendant
			}
			break;

		case 2:
			.@third = eaclass() & EAJ_THIRDMASK;
			switch (.@third) {
				case EAJ_RUNE_KNIGHT:      .@shadow_id = 24698; break; // S_2Para_RK_Weapon
				case EAJ_ROYAL_GUARD:      .@shadow_id = 24699; break; // S_2Para_LG_Weapon
				case EAJ_MECHANIC:         .@shadow_id = 24700; break; // S_2Para_NC_Weapon
				case EAJ_GENETIC:          .@shadow_id = 24701; break; // S_2Para_GN_Weapon
				case EAJ_GUILLOTINE_CROSS: .@shadow_id = 24702; break; // S_2Para_GC_Weapon
				case EAJ_SHADOW_CHASER:    .@shadow_id = 24703; break; // S_2Para_SC_Weapon
				case EAJ_WARLOCK:          .@shadow_id = 24704; break; // S_2Para_WL_Weapon
				case EAJ_SORCERER:         .@shadow_id = 24705; break; // S_2Para_SO_Weapon
				case EAJ_ARCH_BISHOP:      .@shadow_id = 24706; break; // S_2Para_AB_Weapon
				case EAJ_SURA:             .@shadow_id = 24707; break; // S_2Para_SR_Weapon
				case EAJ_RANGER:           .@shadow_id = 24708; break; // S_2Para_RA_Weapon
				case EAJ_MINSTRELWANDERER: .@shadow_id = 24709; break; // S_2Para_WM_Weapon
				case EAJ_KAGEROUOBORO:     .@shadow_id = 24710; break; // S_2Para_KO_Weapon
				case EAJ_REBELLION:        .@shadow_id = 24711; break; // S_2Para_RL_Weapon
				case EAJ_STAR_EMPEROR:     .@shadow_id = 24712; break; // S_2Para_SJ_Weapon
				case EAJ_SOUL_REAPER:      .@shadow_id = 24713; break; // S_2Para_SP_Weapon
				case EAJ_SUMMONER:         .@shadow_id = 24714; break; // S_2Para_Doram_Weapon
				case EAJ_SUPER_NOVICE_E:   .@shadow_id = 24715; break; // S_2Para_NV_Weapon
				default:
					mes "[Lucy]";
					mes "I couldn't identify a verified Third Class profile on you. I cannot issue an advanced weapon.";
					close;
			}
			break;
		}

		if (countitem(25223) < 20) {
			mes "[Lucy]";
			mes "Transaction aborted. You do not have enough Eden Coins.";
			mes "Advanced high-tier rentals require exactly **20 Eden Coins**.";
			close;
		}

		delitem 25223, 20;
		rentitem2 .@shadow_id,259200,1,0,0,0,0,0,0;

		mes "[Lucy]";
		mes "Payment processed. Enjoy your advanced equipment piece!";
		close;

	case 2:
		mes "[Lucy]";
		if (BaseLevel < 10) {
			mes "Access Denied.";
			mes "Beginner Shadow Equipment rentals require you to be at least **Base Level 10**.";
			close;
		}

		mes "These basic shadow items are completely free and last for **72 hours (3 days)**.";
		mes "Which piece would you like?";
		next;

		switch(select("Paradise Shadow Weapon:Paradise Shadow Shield:Paradise Shadow Armor:Paradise Shadow Shoes:Paradise Shadow Earring:Paradise Shadow Pendant")) {
			case 1: .@shadow_id = 24687; break; // S_1Para_Weapon
			case 2: .@shadow_id = 24688; break; // S_1Para_Shield
			case 3: .@shadow_id = 24689; break; // S_1Para_Armor
			case 4: .@shadow_id = 24690; break; // S_1Para_Shoes
			case 5: .@shadow_id = 24691; break; // S_1Para_Earing
			case 6: .@shadow_id = 24692; break; // S_1Para_Pendant
		}

		rentitem2 .@shadow_id,259200,1,0,0,0,0,0,0;
		mes "[Lucy]";
		mes "There you go! Enjoy your free rental piece. It will dissolve in 72 hours.";
		close;

	case 3:
		close;
	}
}
