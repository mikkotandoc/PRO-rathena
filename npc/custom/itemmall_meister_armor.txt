//===== rAthena Script =======================================
//= Item Mall - Meister Basic Armor Vendor
//===== Description: =========================================
//= Sells generic starter armor and Mechanic/Meister job gear
//= on prt_mk (use @mall to warp there).
//============================================================

prt_mk,114,110,4	script	Meister Armorer#prt_mk	4_F_JOB_BLACKSMITH,{
	mes "[Meister Armorer]";
	mes "Need armor for the forge?";
	mes "I carry generic starter pieces and full Mechanic / Meister job sets.";
	next;
	switch (select("Basic Equipment:Mechanic & Meister Gear:Quit")) {
		case 1:
			switch (select("Body Armor:Shields:Footwear & Garments:Accessories:Back")) {
				case 1: callshop "im_mt_body", 1; break;
				case 2: callshop "im_mt_shield", 1; break;
				case 3: callshop "im_mt_feet", 1; break;
				case 4: callshop "im_mt_acc", 1; break;
				case 5: break;
			}
			break;
		case 2:
			switch (select("Body Armor:Garments:Shoes:Shields & Headgear:Back")) {
				case 1: callshop "im_mt_job_body", 1; break;
				case 2: callshop "im_mt_job_garment", 1; break;
				case 3: callshop "im_mt_job_shoes", 1; break;
				case 4: callshop "im_mt_job_misc", 1; break;
				case 5: break;
			}
			break;
		case 3:
			close;
	}
	end;
}
