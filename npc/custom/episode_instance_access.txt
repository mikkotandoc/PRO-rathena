function	script	F_EpClearApply	{
	.@ep = getarg(0);
	.@part = getarg(1, 0);

	switch (.@ep) {
	case 16:
		if (terra_gloria_main < 25)
			terra_gloria_main = 25;
		if (ep16_royal < 24)
			ep16_royal = 24;
		break;
	case 17:
		if (.@part == 1) {
			callfunc("F_EpForceComplete", 11603);
			callfunc("F_EpForceComplete", 16360);
		} else if (.@part == 2) {
			if (ep17_2_main < 36)
				ep17_2_main = 36;
			if (ep172_watergarden < 3)
				ep172_watergarden = 3;
			callfunc("F_EpForceComplete", 18000);
			callfunc("F_EpForceComplete", 18020);
			callfunc("F_EpForceComplete", 12497);
		}
		break;
	case 18:
		if (ep18_main < 57)
			ep18_main = 57;
		#Ep18_Manual_Clear = 1;
		break;
	case 19:
		if (ep19_main < 100)
			ep19_main = 100;
		callfunc("F_EpForceComplete", 17637);
		callfunc("F_EpForceComplete", 18132);
		break;
	case 20:
		if (ep20_main < 100)
			ep20_main = 100;
		callfunc("F_EpForceComplete", 18233);
		callfunc("F_EpForceComplete", 20008);
		break;
	}
	callfunc("F_EpUncloakInstances", .@ep, .@part);
	return;
}

function	script	F_EpUncloakNpcForce	{
	.@name$ = getarg(0);

	if (getnpcid(0, .@name$) == 0)
		return 0;
	cloakoffnpcself(.@name$);
	return 1;
}

function	script	F_Ep17_1InstanceReady	{
	return (isbegin_quest(11603) == 2 || checkquest(17046, PLAYTIME) == 2);
}

function	script	F_EpUncloakEp17_1	{
	.@repair = getarg(0, 0);
	.@fixed = 0;

	if (!callfunc("F_Ep17_1InstanceReady"))
		return 0;

	if (isbegin_quest(12452) == 2) {
		if (.@repair)
			.@fixed += callfunc("F_EpRepairTry", "Operation Officer#171_ocp");
		else
			callfunc("F_EpUncloakNpcForce", "Operation Officer#171_ocp");
	} else {
		if (.@repair)
			.@fixed += callfunc("F_EpRepairTry", "Est#171_ocp");
		else
			callfunc("F_EpUncloakNpcForce", "Est#171_ocp");
	}
	return .@fixed;
}

function	script	F_EpUncloakInstances	{
	.@ep = getarg(0);
	.@part = getarg(1, 0);

	if (.@ep == 16 && terra_gloria_main >= 25)
		callfunc("F_EpUncloakEp16", 0);
	if (.@ep == 17 && .@part == 1 && callfunc("F_Ep17_1InstanceReady"))
		callfunc("F_EpUncloakEp17_1", 0);
	if (.@ep == 17 && .@part == 2 && ep17_2_main >= 1) {
		cloakoffnpcself( "#he_admin1" );
		cloakoffnpcself( "#he_admin2" );
	}
	if (.@ep == 18 && ep18_main >= 52) {
		cloakoffnpcself( "Priest#sms01" );
		if (ep18_main >= 54) {
			cloakoffnpcself( "Ira#c12_01" );
			cloakoffnpcself( "Miriam#c12_01" );
			cloakoffnpcself( "Maram#c12_01" );
			cloakoffnpcself( "Suad#c12_01" );
		}
		if (ep18_main >= 57)
			cloakoffnpcself( "Aira#con" );
	}
	if (.@ep == 19 && ep19_main >= 50) {
		cloakoffnpcself( "Juncea#ep19re2" );
		cloakoffnpcself( "#warp_ep19re1" );
	}
	if (.@ep == 19 && ep19_main >= 59) {
		cloakoffnpcself( "Lehar#i19ms21" );
		cloakoffnpcself( "Lazy#i19ms21" );
		cloakoffnpcself( "Aurelie#i19ms21" );
		cloakoffnpcself( "Horuru#i19ms21" );
		cloakoffnpcself( "Horr#i19ms21" );
		cloakoffnpcself( "Voglinde#i19ms21" );
		cloakoffnpcself( "Leon#i19ms21" );
		cloakoffnpcself( "Miriam#i19ms21" );
		cloakoffnpcself( "Iwin#i19ms21" );
		cloakoffnpcself( "Iwin#i19ms22" );
	}
	if (.@ep == 19 && ep19_main >= 100)
		cloakoffnpcself( "Arolong#ep19re2" );
	if (.@ep == 20 && ep20_main >= 13)
		cloakoffnpcself( "Canyon Exploration#ep20" );
	if (.@ep == 20 && ep20_main >= 21)
		cloakoffnpcself( "Drift Ice Zone#ep20" );
	if (.@ep == 20 && ep20_main >= 25)
		cloakoffnpcself( "Nest of Twigs#ep20" );
	if (.@ep == 20 && ep20_main >= 29)
		cloakoffnpcself( "The Immortal#ep20" );
	return;
}

function	script	F_EpRepairTry	{
	.@name$ = getarg(0);

	if (getnpcid(0, .@name$) == 0)
		return 0;
	if (!isnpccloaked(.@name$))
		return 0;
	cloakoffnpcself(.@name$);
	@ep_repair_names$[getarraysize(@ep_repair_names$)] = .@name$;
	return 1;
}

			callfunc("F_EpUncloakEp16", 0);

	if (.@map$ == "sp_cor" && callfunc("F_Ep17_1InstanceReady"))
		callfunc("F_EpUncloakEp17_1", 0);
