//===== rAthena Script =======================================
//= Instance Cooldown Eraser
//===== Description: =========================================
//= Zeny sink NPC that clears memorial dungeon waiting timers.
//= Covers all instances that use PLAYTIME quest cooldowns.
//===== Pricing Tiers (zeny) =================================
//= Tier 1 - Short / re-entry (<=2h):     225,000
//= Tier 2 - Standard daily (4h):          787,500
//= Tier 3 - Extended (12h-1d):          1,575,000
//= Tier 4 - Multi-day (2-4d):           3,375,000
//= Tier 5 - Weekly (7d):                6,750,000
//= Tier 6 - Rare weekly lock:          11,250,000
//============================================================

function	script	F_InstCD_Active	{
	.@qid = getarg(0);
	.@st = checkquest(.@qid, PLAYTIME);
	return (.@st == 0 || .@st == 1);
}

function	script	F_InstCD_Clear	{
	.@qid = getarg(0);
	if (checkquest(.@qid) > -1)
		erasequest .@qid;
	return;
}

	function	IsOnCooldown	{
		.@idx = getarg(0);
		.@start = .qoff[.@idx];
		.@end = .@start + .qnum[.@idx];
		for (.@q = .@start; .@q < .@end; .@q++) {
			if (callfunc("F_InstCD_Active", .quests[.@q]))
				return 1;
		}
		return 0;
	}

	function	ClearEntry	{
		.@idx = getarg(0);
		.@start = .qoff[.@idx];
		.@end = .@start + .qnum[.@idx];
		for (.@q = .@start; .@q < .@end; .@q++)
			callfunc("F_InstCD_Clear", .quests[.@q]);
		return;
	}

	// getarg(0) = list size, getarg(1) = 0 browse list / 1 active cooldown list
	// Returns instance index, or -1 when cancelled.
	function	SelectEntry	{
		.@list_size = getarg(0);
		.@active_list = getarg(1, 0);
		.@page = 0;

		while (1) {
			.@start = .@page * .page_size;
			if (.@start >= .@list_size) {
				.@page = 0;
				.@start = 0;
			}
			.@end = .@start + .page_size;
			if (.@end > .@list_size)
				.@end = .@list_size;
			.@count = .@end - .@start;

			if (.@list_size > .page_size) {
				.@pages = (.@list_size + .page_size - 1) / .page_size;
				mes "[Instance Eraser]";
				mes "Select a memorial dungeon. (Page " + (.@page + 1) + " / " + .@pages + ")";
				next;
			}

			.@menu$ = "";
			for (.@i = .@start; .@i < .@end; .@i++) {
				if (.@active_list)
					.@menu$ = .@menu$ + .menu_active_name$[.@i] + ":";
				else
					.@menu$ = .@menu$ + .name$[ .menu_idx[.@i] ] + ":";
			}

			.@has_prev = (.@page > 0);
			.@has_next = (.@end < .@list_size);
			if (.@has_prev)
				.@menu$ = .@menu$ + "Previous Page:";
			if (.@has_next)
				.@menu$ = .@menu$ + "Next Page:";
			.@menu$ = .@menu$ + "Cancel";

			.@pick = select(.@menu$) - 1;
			if (.@pick < .@count) {
				if (.@active_list)
					return .menu_active_idx[.@start + .@pick];
				return .menu_idx[.@start + .@pick];
			}
			.@pick = .@pick - .@count;

			if (.@has_prev) {
				if (.@pick == 0) {
					.@page--;
					continue;
				}
				.@pick--;
			}
			if (.@has_next && .@pick == 0) {
				.@page++;
				continue;
			}
			return -1;
		}
	}

	function	BuildMenu	{
		.@show_all = getarg(0);
		deletearray .menu_idx;
		deletearray .menu_active_idx;
		deletearray .menu_active_name$;
		deletearray .menu_active_cost;
		.menu_active = 0;
		.menu_total = 0;
		.@shown = 0;

		for (.@i = 0; .@i < getarraysize(.name$); .@i++) {
			.@on = IsOnCooldown(.@i);
			if (!.@show_all && !.@on)
				continue;

			.menu_idx[.@shown] = .@i;

			if (.@on) {
				.menu_active_idx[.menu_active] = .@i;
				.menu_active_name$[.menu_active] = .name$[.@i];
				.menu_active_cost[.menu_active] = .cost[.@i];
				.menu_total += .cost[.@i];
				.menu_active++;
			}
			.@shown++;
		}
		return .@shown;
	}
