//===== rAthena Script =======================================
//= Varmundt's Biosphere Depth - Level 2 Daily Hunt
//===== Description: =========================================
//= Depth 2: 450 kills in bl_depth2 (Base Level 250+).
//= Depth 1 uses official quests in npc/re/quests/biosphere_depth1.txt.
//============================================================

ba_chess,25,13,5	script	Depth Surveyor#bs_d2	4_EP17_BASIC_B,{
	if (BaseLevel < 250) {
		mes "[Depth Surveyor]";
		mes "The abyss survey is only available to adventurers with base level 250 or higher.";
		close;
	}
	if (isbegin_quest(16734) != 2) {
		mes "[Depth Surveyor]";
		mes "You must complete the Depth Biosphere introduction with the Depth Research Manager first.";
		mes "^777777(ba_in01 292, 104)^000000";
		close;
	}
	mes "[Depth Surveyor]";
	mes "Would you like to take on today's depth abyss hunt?";
	mes "Eliminate ^0000cd450^000000 monsters inside depth level 2.";
	next;
	if (select("Accept daily hunt", "Cancel") == 2) {
		mes "[Depth Surveyor]";
		mes "Return when you are prepared.";
		close;
	}
	callsub F_DepthDaily, 16779, 16780, "Depth Level 2", "bl_depth2", 450, "Time_D_Ma_Rune", 1000000000, 600000000;
	end;

F_DepthDaily:
	.@hunt = getarg(0);
	.@wait = getarg(1);
	.@name$ = getarg(2);
	.@reward$ = getarg(5);
	.@bexp = getarg(6);
	.@jexp = getarg(7);

	switch (checkquest(.@wait, PLAYTIME)) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Depth Surveyor]";
		mes "You have already completed the ^0000cd" + .@name$ + "^000000 daily hunt today.";
		close;
	case 2:
		erasequest .@wait;
		break;
	}

	switch (checkquest(.@hunt, HUNTING)) {
	case -1:
		setquest .@hunt;
		mes "[Depth Surveyor]";
		mes "Good luck. Any monster in ^0000cd" + .@name$ + "^000000 counts toward the goal.";
		close;
	case 0:
	case 1:
		mes "[Depth Surveyor]";
		mes "Keep hunting in ^0000cd" + .@name$ + "^000000 until you reach the target.";
		close;
	case 2:
		erasequest .@hunt;
		setquest .@wait;
		getitem .@reward$, 1;
		getexp .@bexp, .@jexp;
		mes "[Depth Surveyor]";
		mes "Survey complete. Here is your reward.";
		close;
	}
	end;

OnInit:
	setunittitle getnpcid(0), "<Depth 2 Daily>";
	end;
}
