// Eden Coin exchange at HQ (same shop as Paradise Market)
moc_para01,108,35,4	script	Eden Coin Trader#hq	4_F_KAFRA7,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "[Eden Coin Trader]";
		mes "The Eden Group Coin exchange is for registered members only.";
		close;
	}

	mes "[Eden Coin Trader]";
	mes "Trade Eden Group Coins earned from missions for supplies.";
	mes "A second counter is also open at the Paradise Market.";
	next;

	if (select("Open Exchange Shop:Cancel") == 2) {
		mes "[Eden Coin Trader]";
		mes "Come back when you have coins to spend.";
		close;
	}

	if (!getbattleflag("feature.barter")) {
		mes "[Eden Coin Trader]";
		mes "The barter system is currently disabled.";
		close;
	}

	close2;
	callshop "para_edencoin10", 1;
	end;
}
