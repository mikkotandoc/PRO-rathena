//============================================================
// Eden Group Event Quest Board
// Daily hunting events rewarding Eden Group Coins (25223)
// Quest IDs 70289 ~ 70300 (client: OngoingQuests.lub)
// Hunt quests 70289 ~ 70300 (+24h to complete)
// Cooldown IDs 70309 ~ 70320 (+168h / 7 days after turn-in)
//============================================================

moc_para01,33,35,3	script	Eden Event Coordinator	4_F_KAFRA8,{

	mes "[Eden Event Coordinator]";
	mes "Welcome! I coordinate limited-time hunting events for Eden members.";
	mes "Complete assignments to earn EXP and Eden Group Coins.";
	next;

	if (countitem(6219) == 0 && countitem(22508) == 0) {
		mes "[Eden Event Coordinator]";
		mes "These special event assignments are reserved for registered Eden Group members.";
		close;
	}

	switch(select("Browse Event Quests:Turn In Completed Quests:Cancel")) {
	case 1:
		callsub L_Cleanup;
		mes "[Eden Event Coordinator]";
		mes "Select an event assignment:";
		mes "Each hunt must be finished within ^0000cd24 hours^000000.";
		mes "After turn-in, that event enters a ^0000cd7-day cooldown^000000 before you can take it again.";
		next;
		switch(select("[Event] Poring Extermination:[Event] Ant Egg Collection:[Event] Skeleton Warriors:[Event] Orc Zombie Cleanup:[Event] Mi Gao Extermination:[Event] Anubis Guardian Trial:[Event] Illusion of Moonlight:[Event] Illusion of Vampire:[Event] Illusion of Teddy Bear:[Event] Illusion of Abyss:[Event] Illusion of Twins:[Event] Illusion of Labyrinth:Cancel")) {
			case 1:  callsub L_Assign, 70289, 70309, "Poring Extermination"; break;
			case 2:  callsub L_Assign, 70290, 70310, "Ant Egg Collection"; break;
			case 3:  callsub L_Assign, 70291, 70311, "Skeleton Warriors"; break;
			case 4:  callsub L_Assign, 70292, 70312, "Orc Zombie Cleanup"; break;
			case 5:  callsub L_Assign, 70293, 70313, "Mi Gao Extermination"; break;
			case 6:  callsub L_Assign, 70294, 70314, "Anubis Guardian Trial"; break;
			case 7:  callsub L_Assign, 70295, 70315, "Illusion of Moonlight"; break;
			case 8:  callsub L_Assign, 70296, 70316, "Illusion of Vampire"; break;
			case 9:  callsub L_Assign, 70297, 70317, "Illusion of Teddy Bear"; break;
			case 10: callsub L_Assign, 70298, 70318, "Illusion of Abyss"; break;
			case 11: callsub L_Assign, 70299, 70319, "Illusion of Twins"; break;
			case 12: callsub L_Assign, 70300, 70320, "Illusion of Labyrinth"; break;
			case 13: close;
		}
		break;

	case 2:
		if (checkweight(1201,1) == 0 || (MaxWeight - Weight) < 2000) {
			mes "[Eden Event Coordinator]";
			mes "You are carrying too much to receive event rewards.";
			mes "Please free up inventory space and try again.";
			close;
		}

		callsub L_Cleanup;
		mes "[Eden Event Coordinator]";
		mes "Checking your completed event assignments...";
		next;

		for (.@i = 70289; .@i <= 70300; .@i++) {
			if (checkquest(.@i) < 0)
				continue;
			if (checkquest(.@i, HUNTING) != 2)
				continue;

			.@cd_id = .@i + 20;
			callsub L_Reward, .@i, .@cd_id;
			.@rewarded = 1;
		}

		if (!.@rewarded) {
			mes "[Eden Event Coordinator]";
			mes "You do not have any completed event quests to turn in.";
			close;
		}

		mes "[Eden Event Coordinator]";
		mes "Your completed event quests have been rewarded.";
		mes "Each turned-in assignment is now on a ^0000cd7-day cooldown^000000.";
		mes "You may repeat it after the timer shown in your Quest Window expires.";
		close;

	case 3:
		close;
	}
	end;

L_Cleanup:
	for (.@i = 70309; .@i <= 70320; .@i++) {
		if (checkquest(.@i) < 0)
			continue;
		if (checkquest(.@i, PLAYTIME) == 2)
			erasequest .@i;
	}
	for (.@i = 70289; .@i <= 70300; .@i++) {
		if (checkquest(.@i) < 0)
			continue;
		if (checkquest(.@i, PLAYTIME) == 2)
			erasequest .@i;
	}
	return;

L_Assign:
	.@q_id = getarg(0);
	.@cd_id = getarg(1);
	.@name$ = getarg(2);

	if (checkquest(.@cd_id, PLAYTIME) == 0 || checkquest(.@cd_id, PLAYTIME) == 1) {
		mes "[Eden Event Coordinator]";
		mes "^ff0000" + .@name$ + "^000000 is on a ^0000cd7-day cooldown^000000.";
		mes "Check your Quest Window for the remaining wait time, then try again.";
		close;
	}
	if (checkquest(.@cd_id, PLAYTIME) == 2)
		erasequest .@cd_id;

	if (checkquest(.@q_id) >= 0 && checkquest(.@q_id, PLAYTIME) == 2) {
		erasequest .@q_id;
		mes "[Eden Event Coordinator]";
		mes "Your previous ^0000ff" + .@name$ + "^000000 attempt expired after ^0000cd24 hours^000000 and has been reset.";
		next;
	}

	if (checkquest(.@q_id) < 0) {
		setquest .@q_id;
		mes "[Eden Event Coordinator]";
		mes "Event registered: ^0000ff" + .@name$ + "^000000.";
		mes "You have ^0000cd24 hours^000000 to complete this hunt.";
		mes "Return here to turn it in once your Quest Window shows it as finished.";
		close;
	}

	switch (checkquest(.@q_id, HUNTING)) {
	case -1:
		erasequest .@q_id;
		break;
	case 2:
		mes "[Eden Event Coordinator]";
		mes "You have already completed ^0000ff" + .@name$ + "^000000.";
		mes "Please select ^0000ffTurn In Completed Quests^000000 to collect your reward.";
		close;
	case 0:
	case 1:
		mes "[Eden Event Coordinator]";
		mes "You are already tracking ^0000ff" + .@name$ + "^000000.";
		mes "Finish the hunt within ^0000cd24 hours^000000, or wait for an expired attempt to reset.";
		close;
	}

	setquest .@q_id;
	mes "[Eden Event Coordinator]";
	mes "Event registered: ^0000ff" + .@name$ + "^000000.";
	mes "You have ^0000cd24 hours^000000 to complete this hunt.";
	mes "Return here to turn it in once your Quest Window shows it as finished.";
	close;

L_Reward:
	.@q_id = getarg(0);
	.@cd_id = getarg(1);

	switch (.@q_id) {
		case 70289: .@b_exp = 3000;  .@j_exp = 2000;  .@rate = 200; .@coin = 10; break;
		case 70290: .@b_exp = 3200;  .@j_exp = 3000;  .@rate = 200; .@coin = 15; break;
		case 70291: .@b_exp = 14350; .@j_exp = 13700; .@rate = 220; .@coin = 20; break;
		case 70292: .@b_exp = 14350; .@j_exp = 13700; .@rate = 220; .@coin = 25; break;
		case 70293: .@b_exp = 39600; .@j_exp = 35000; .@rate = 220; .@coin = 30; break;
		case 70294: .@b_exp = 72000; .@j_exp = 72000; .@rate = 220; .@coin = 40; break;
		case 70295: .@b_exp = 160000; .@j_exp = 160000; .@rate = 150; .@coin = 50; break;
		case 70296: .@b_exp = 300000; .@j_exp = 300000; .@rate = 150; .@coin = 60; break;
		case 70297: .@b_exp = 800000;  .@j_exp = 800000;  .@rate = 200; .@coin = 60; break;
		case 70298: .@b_exp = 1000000; .@j_exp = 1000000; .@rate = 200; .@coin = 75; break;
		case 70299: .@b_exp = 1110729; .@j_exp = 1413414; .@rate = 200; .@coin = 75; break;
		case 70300: .@b_exp = 1042488; .@j_exp = 2760588; .@rate = 200; .@coin = 75; break;
		default:
			return;
	}

	.@b_exp = (.@b_exp * 80 * getbattleflag("base_exp_rate")) / (.@rate * 100);
	.@j_exp = (.@j_exp * 80 * getbattleflag("job_exp_rate")) / (.@rate * 100);

	erasequest .@q_id;

	if (checkquest(.@cd_id) >= 0)
		erasequest .@cd_id;

	setquest .@cd_id;
	getexp .@b_exp, .@j_exp;
	getitem 25223, .@coin;
	return;
}
