// Player ping command - type @ping in chat to check your latency.
-	script	ping_cmd	-1,{
OnCall:
	.@ms = getping();
	if( .@ms < 0 ){
		requestping();
		sleep2 300;
		.@ms = getping();
	}
	if( .@ms < 0 )
		dispbottom "Unable to measure your latency. Please try again.";
	else
		dispbottom "Your current latency is: " + .@ms + " ms.";
	end;

OnInit:
	bindatcmd "ping", strnpcinfo(0)+"::OnCall", 0, 99;
	end;
}
