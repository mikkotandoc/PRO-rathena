//============================================================

// Automatic macro detector patrol

// Only patrols maps that drop Play_RO_Gold_Coin_ (db/re/map_drops.yml).

// Regenerate the farm map list: powershell tools/sync_macro_patrol_maps.ps1

//============================================================

-	script	MacroPatrol	-1,{

	end;



OnInit:

	.interval_ms = 60000;    // TEST: 1 minute (restore to 600000 for 10 minutes)

	.max_checks = 5;          // Players challenged per cycle

	.min_baselevel = 30;     // Skip low-level/new chars

	.player_cooldown_ms = 1800000; // 30 minutes before same char is checked again

	.gm_skip_level = 1;       // Skip staff at or above this GM level

	.enabled = 1;              // Automatic patrol on/off
	.last_run_at = 0;          // gettimetick(2) of last patrol cycle
	.last_players = 0;         // Players on farm maps (last cycle)
	.last_eligible = 0;        // Eligible targets (last cycle)
	.last_checked = 0;         // Macro checks started (last cycle)
	.report_aid = 0;           // Account id to notify after manual run
	.manual_run = 0;           // 1 when invoked by @macropatrol run

	.current_cycle_id = 0;    // SQL macro_patrol_cycle.id for the active patrol run

	.log_enabled = 1;          // Write patrol/challenge rows to macro_patrol_* tables

	// Maps with Play_RO_Gold_Coin_ in GlobalDrops (db/re + db/import map_drops.yml)
	// Regenerate: powershell tools/sync_macro_patrol_maps.ps1  (or node tools/sync_macro_patrol_maps.js)
	// EP19 farmable: jor_back1-3, jor_tail, jor_ab01-02, jor_dun01-02
	// EP20 farmable: jor_back4-6, jor_maze, jor_root1-3, jor_twice, jor_twig
	// EP19.x farmable: mjo_wst01, ra_pol01

	setarray .farm_maps$,
		"abyss_01", "abyss_02", "abyss_03", "abyss_04",
		"amicitia1", "amicitia2", "ant_d02_i",
		"ba_bath", "ba_lib", "ba_maison",
		"bif_fild01", "bif_fild02",
		"bl_death", "bl_depth1", "bl_depth2", "bl_grass", "bl_ice", "bl_lava", "bl_soul", "bl_temple", "bl_venom",
		"clock_01", "com_d02_i",
		"dic_fild01", "dic_fild02", "ecl_fild01", "ein_d02_i", "ein_dun03",
		"gef_d01_i", "gl_prison1", "gw_fild01", "gw_fild02",
		"ice_d03_i", "ice_dun03", "iz_d04_i", "iz_d05_i",
		"jor_ab01", "jor_ab02", "jor_albe", "jor_back1", "jor_back2", "jor_back3", "jor_back4", "jor_back5", "jor_back6",
		"jor_base", "jor_crk", "jor_crk_p", "jor_dun01", "jor_dun02", "jor_dun03", "jor_maze", "jor_mbase", "jor_nest",
		"jor_que", "jor_raise1", "jor_raise2", "jor_root1", "jor_root2", "jor_root3", "jor_safty1", "jor_safty2", "jor_sanct",
		"jor_sklf1", "jor_sklf2", "jor_tail", "jor_tmple1", "jor_tmple2", "jor_twice", "jor_twig",
		"juperos_01", "kh_dun01", "kh_dun02", "lasa_dun02", "lhz_dun02", "lhz_dun03", "lhz_dun04", "lou_dun03", "mag_dun02",
		"mjo_wst01",
		"nif_dun01", "nif_dun02", "odin_past", "odin_tem02", "odin_tem03",
		"oz_dun01", "oz_dun02", "pay_d03_i", "prt_mz03_i",
		"ra_fild01", "ra_fild10", "ra_fild11", "ra_fild12", "ra_fild13", "ra_pol01",
		"sp_rudus", "sp_rudus2", "sp_rudus3", "sp_rudus4", "spl_fild02",
		"tha_t01", "tha_t02", "tha_t03", "tha_t04", "tha_t05", "tha_t06",
		"tur_d03_i", "tur_d04_i", "tur_d04ia", "tur_d04ib", "ve_fild03";



	for (.@i = 0; .@i < getarraysize(.farm_maps$); .@i++)

		removemapflag .farm_maps$[.@i], mf_nomacrochecker;



	bindatcmd "macropatrol", strnpcinfo(0)+"::OnAtcommand", 1, 99;

	initnpctimer;

	end;



OnAtcommand:

	.@cmd$ = "help";

	if (.@atcmd_numparameters > 0)

		.@cmd$ = .@atcmd_parameters$[0];



	if (.@cmd$ == "help") {

		dispbottom "=== Macro Patrol ===";

		dispbottom "@macropatrol status  - patrol config and last run stats";

		dispbottom "@macropatrol run    - run one patrol cycle now";

		dispbottom "@macropatrol test    - trigger captcha check on yourself";

		dispbottom "@macropatrol on      - enable automatic patrol timer";

		dispbottom "@macropatrol off     - disable automatic patrol timer";

		end;

	}



	if (.@cmd$ == "status") {

		callsub(OnShowStatus);

		end;

	}



	if (.@cmd$ == "on") {

		if (.enabled) {

			dispbottom "Macro Patrol: automatic patrol is already enabled.";

			end;

		}

		.enabled = 1;

		initnpctimer;

		dispbottom "Macro Patrol: automatic patrol enabled (every " + (.interval_ms / 60000) + " min).";

		end;

	}



	if (.@cmd$ == "off") {

		if (!.enabled) {

			dispbottom "Macro Patrol: automatic patrol is already disabled.";

			end;

		}

		.enabled = 0;

		stopnpctimer;

		dispbottom "Macro Patrol: automatic patrol disabled. Use @macropatrol run to trigger manually.";

		end;

	}



	if (.@cmd$ == "test") {

		if (getbattleflag("macro_detection_punishment") != 1) {

			dispbottom "Macro Patrol: macro_detection_punishment must be jail (1). Patrol is inactive.";

			end;

		}

		dispbottom "Macro Patrol: starting captcha test on you.";

		dispbottom "Requires a patched client (Nemo: Fix captcha decompression buffer size).";

		dispbottom "If the client crashes, apply that patch to proasia.exe then retry.";

		dispbottom "Answer is case-sensitive 4-letter word (e.g. WOLF, BEAR). Fail/timeout = jail.";

		callsub(OnStartChallenge, "manual_test");

		macro_detector getcharid(3);

		end;

	}



	if (.@cmd$ == "run") {

		.manual_run = 1;

		.report_aid = getcharid(3);

		donpcevent strnpcinfo(0)+"::OnPatrol";

		end;

	}



	dispbottom "Unknown option. Use @macropatrol help";

	end;



OnShowStatus:

	.@caller_aid = getcharid(3);

	.@farm_count = getarraysize(.farm_maps$);

	.@players = 0;

	.@eligible = 0;



	for (.@m = 0; .@m < .@farm_count; .@m++) {

		.@found = getmapunits(BL_PC, .farm_maps$[.@m], .@aid[.@players]);

		.@players += .@found;

	}



	for (.@i = 0; .@i < .@players; .@i++) {

		if (.@aid[.@i] <= 0)

			continue;

		if (!attachrid(.@aid[.@i])) {

			detachrid;

			continue;

		}

		if (callsub(OnEligible) == 1)

			.@eligible++;

		detachrid;

	}



	if (!attachrid(.@caller_aid))

		return;



	dispbottom "=== Macro Patrol Status ===";

	dispbottom "Automatic patrol: " + (.enabled ? "ON" : "OFF") + " (interval " + (.interval_ms / 60000) + " min, max " + .max_checks + "/cycle)";

	dispbottom "Farm maps: " + .@farm_count + " | Players on farms: " + .@players + " | Eligible now: " + .@eligible;

	dispbottom "Filters: base lv >= " + .min_baselevel + ", GM skip >= " + .gm_skip_level + ", cooldown " + (.player_cooldown_ms / 60000) + " min";

	if (getbattleflag("macro_detection_punishment") == 1)

		dispbottom "Punishment: jail " + getbattleflag("macro_detection_punishment_time") + " min | retry " + getbattleflag("macro_detection_retry") + " | timeout " + (getbattleflag("macro_detection_timeout") / 1000) + "s";

	else

		dispbottom "Punishment: NOT jail (macro_detection_punishment != 1) — patrol will not run.";



	if (.last_run_at > 0) {

		.@ago = gettimetick(2) - .last_run_at;

		dispbottom "Last cycle: " + (.@ago / 1000) + "s ago | players " + .last_players + " | eligible " + .last_eligible + " | checked " + .last_checked;

	} 	else

		dispbottom "Last cycle: never (or since last @reloadscript).";



	dispbottom "SQL logging: " + (.log_enabled ? "ON" : "OFF");



	if (compare(strcharinfo(3), "") && inarray(.farm_maps$, strcharinfo(3)) >= 0) {

		if (#macro_patrol_next > gettimetick(2))

			dispbottom "Your cooldown: " + ((#macro_patrol_next - gettimetick(2)) / 1000) + "s remaining";

		else

			dispbottom "Your cooldown: none (eligible if other filters pass)";

	}

	return;



OnReport:

	if (.report_aid <= 0)

		return;



	if (!attachrid(.report_aid)) {

		.report_aid = 0;

		.manual_run = 0;

		return;

	}



	if (getbattleflag("macro_detection_punishment") != 1)

		dispbottom "Macro Patrol: patrol skipped — macro_detection_punishment is not jail (1).";

	else if (.last_players <= 0)

		dispbottom "Macro Patrol: no players on farm maps.";

	else if (.last_checked <= 0)

		dispbottom "Macro Patrol: " + .last_players + " on farm maps, " + .last_eligible + " eligible, but no checks started.";

	else

		dispbottom "Macro Patrol: started " + .last_checked + " check(s) (" + .last_eligible + " eligible / " + .last_players + " on farms).";



	detachrid;

	.report_aid = 0;

	.manual_run = 0;

	return;



OnTimer60000:

	if (!.enabled) {

		initnpctimer;

		end;

	}



	donpcevent strnpcinfo(0)+"::OnPatrol";

	initnpctimer;

	end;



OnPatrol:

	if (!.manual_run && !.enabled)

		end;



	if (getbattleflag("macro_detection_punishment") != 1) {

		debugmes "MacroPatrol: macro_detection_punishment is not set to jail (1).";

		callsub(OnReport);

		end;

	}



	.@player_count = 0;

	.@eligible_count = 0;

	for (.@m = 0; .@m < getarraysize(.farm_maps$); .@m++) {

		.@found = getmapunits(BL_PC, .farm_maps$[.@m], .@aid[.@player_count]);

		.@player_count += .@found;

	}



	for (.@i = 0; .@i < .@player_count; .@i++) {

		if (.@aid[.@i] <= 0)

			continue;

		if (!attachrid(.@aid[.@i])) {

			detachrid;

			continue;

		}

		if (callsub(OnEligible) == 1)

			.@eligible_count++;

		detachrid;

	}



	.last_run_at = gettimetick(2);

	.last_players = .@player_count;

	.last_eligible = .@eligible_count;

	.last_checked = 0;



	if (.log_enabled)

		callsub(OnLogCycleStart, .@player_count, .@eligible_count);



	if (.@player_count <= 0) {

		callsub(OnReport);

		end;

	}



	.@checked = 0;

	.@attempts = 0;

	.@max_attempts = .@player_count * 3;



	while (.@checked < .max_checks && .@attempts < .@max_attempts) {

		.@attempts++;

		if (.@player_count <= 1)

			.@idx = 0;

		else

			.@idx = rand(.@player_count);

		.@target_aid = .@aid[.@idx];



		if (.@target_aid <= 0)

			continue;



		if (!attachrid(.@target_aid)) {

			detachrid;

			continue;

		}



		if (callsub(OnEligible) == 0) {

			detachrid;

			continue;

		}



		#macro_patrol_next = gettimetick(2) + .player_cooldown_ms;

		callsub(OnStartChallenge, "patrol");

		macro_detector getcharid(3);

		detachrid;



		.@checked++;

	}



	.last_checked = .@checked;



	if (.log_enabled)

		callsub(OnLogCycleFinish);



	if (.@checked > 0)

		debugmes "MacroPatrol: started macro detection on " + .@checked + " player(s) on Play_RO_Gold_Coin_ farming maps.";



	callsub(OnReport);

	end;



OnEligible:

	if (getgmlevel() >= .gm_skip_level)

		return 0;



	if (BaseLevel < .min_baselevel)

		return 0;



	if (strcharinfo(3) == "sec_pri")

		return 0;



	if (getmapflag(strcharinfo(3), mf_nomacrochecker))

		return 0;



	if (getmapflag(strcharinfo(3), mf_town))

		return 0;



	if (getmapflag(strcharinfo(3), mf_battleground))

		return 0;



	if (compare(strcharinfo(3), "@"))

		return 0;



	if (inarray(.farm_maps$, strcharinfo(3)) < 0)

		return 0;



	if (checkvending(strcharinfo(0)) == 2)

		return 0;



	if (#macro_patrol_next > gettimetick(2))

		return 0;



	return 1;



OnLogCycleStart:

	.@players = getarg(0);

	.@eligible = getarg(1);

	query_sql("INSERT INTO `macro_patrol_cycle` (`cycle_time`,`players_on_farm`,`eligible`,`manual_run`) VALUES (NOW()," + .@players + "," + .@eligible + "," + .manual_run + ")");

	query_sql("SELECT LAST_INSERT_ID()", .@cycle_id);

	if (getarraysize(.@cycle_id) > 0)

		.current_cycle_id = .@cycle_id[0];

	return;



OnLogCycleFinish:

	if (.current_cycle_id <= 0)

		return;

	query_sql("UPDATE `macro_patrol_cycle` SET `challenged` = " + .last_checked + " WHERE `id` = " + .current_cycle_id);

	return;



OnStartChallenge:

	.@source$ = getarg(0);

	#macro_patrol_active = 1;

	#macro_patrol_challenge_id = 0;

	if (!.log_enabled)

		return;

	.@aid = getcharid(3);

	.@cid = getcharid(0);

	.@name$ = escape_sql(strcharinfo(0));

	.@map$ = escape_sql(strcharinfo(3));

	query_sql("INSERT INTO `macro_patrol_challenge` (`cycle_id`,`challenge_time`,`account_id`,`char_id`,`char_name`,`map`,`base_level`,`job_level`,`source`) VALUES (" + .current_cycle_id + ",NOW()," + .@aid + "," + .@cid + ",'" + .@name$ + "','" + .@map$ + "'," + BaseLevel + "," + JobLevel + ",'" + escape_sql(.@source$) + "')");

	query_sql("SELECT LAST_INSERT_ID()", .@log_id);

	if (getarraysize(.@log_id) > 0)

		#macro_patrol_challenge_id = .@log_id[0];

	return;



OnMacroDetectFail:

	if (#macro_patrol_active <= 0)

		end;

	.@fail = #macro_patrol_fail_type;

	.@result$ = "timeout";

	if (.@fail == 1)

		.@result$ = "disconnect";

	else if (.@fail == 3)

		.@result$ = "incorrect";

	if (.log_enabled && #macro_patrol_challenge_id > 0)

		query_sql("UPDATE `macro_patrol_challenge` SET `result` = '" + escape_sql(.@result$) + "', `result_time` = NOW(), `jail_minutes` = " + getbattleflag("macro_detection_punishment_time") + " WHERE `id` = " + #macro_patrol_challenge_id + " AND `result` = 'pending'");

	#macro_patrol_active = 0;

	#macro_patrol_challenge_id = 0;

	#macro_patrol_fail_type = 0;

	end;



OnMacroDetectPass:

	if (#macro_patrol_active <= 0)

		end;

	if (.log_enabled && #macro_patrol_challenge_id > 0)

		query_sql("UPDATE `macro_patrol_challenge` SET `result` = 'passed', `result_time` = NOW() WHERE `id` = " + #macro_patrol_challenge_id + " AND `result` = 'pending'");

	#macro_patrol_active = 0;

	#macro_patrol_challenge_id = 0;

	#macro_patrol_fail_type = 0;

	end;



}
