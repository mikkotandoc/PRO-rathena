prontera,153,191,4	script	Silvervine Quester	4_M_MERCAT1,{

	// ==========================================
	// DIALOGUE CORE LOGIC
	// ==========================================
	mes "[Silvervine Quester]";
	mes "Greetings! I hand out rewards to players who can take down elusive mini-bosses.";
	next;

	// Daily Limit Check & Reset
	if (#Silvervine_LastDate != gettime(DT_YYYYMMDD)) {
		#Silvervine_LastDate = gettime(DT_YYYYMMDD);
		#Silvervine_DailyCount = 0;
	}

	// Check if the player currently has an assigned hunt target stored
	if (SilvQuest_ID > 0) {
		.@known = 0;
		for (.@i = 0; .@i < .quest_count; .@i++) {
			if (.quest_id[.@i] == SilvQuest_ID) {
				.@known = 1;
				break;
			}
		}
		if (!.@known) {
			if (isbegin_quest(SilvQuest_ID))
				erasequest SilvQuest_ID;
			SilvQuest_ID = 0;
		}
	}

	if (SilvQuest_ID > 0) {
		.@hunt_status = checkquest(SilvQuest_ID, HUNTING);

		if (.@hunt_status == 2) {
			mes "[Silvervine Quester]";
			mes "Superb work! You successfully hunted down the target.";

			erasequest SilvQuest_ID;
			SilvQuest_ID = 0;

			getitem .reward_item, .reward_amount;
			#Silvervine_DailyCount = #Silvervine_DailyCount + 1;

			mes "Received: ^00FF00" + .reward_amount + "x Silvervine Fruit^000000";
			mes "Completed Today: ^00FF00" + #Silvervine_DailyCount + "/" + .max_daily_quests + "^000000";
			close;
		}

		.@active_name$ = "Unknown Target";
		for (.@i = 0; .@i < .quest_count; .@i++) {
			if (.quest_id[.@i] == SilvQuest_ID) {
				.@active_name$ = getmonsterinfo(.quest_mob$[.@i], MOB_NAME);
				break;
			}
		}

		mes "[Silvervine Quester]";
		mes "Your current target is: ^FF0000" + .@active_name$ + "^000000.";
		mes "Please eliminate them and check back with me.";
		close;
	}

	// Enforce daily hard ceiling limit
	if (#Silvervine_DailyCount >= .max_daily_quests) {
		mes "[Silvervine Quester]";
		mes "You have reached your limit of ^FF0000" + .max_daily_quests + " hunts^000000 for today.";
		mes "Come back tomorrow after midnight server time!";
		close;
	}

	// Roll a new contract
	mes "[Silvervine Quester]";
	mes "Are you ready to hunt a random mini-boss?";
	mes "Progress today: " + #Silvervine_DailyCount + " / " + .max_daily_quests;
	next;

	if (select("Give me a target!:Maybe later.") == 2) {
		mes "[Silvervine Quester]";
		mes "Be safe out there.";
		close;
	}

	.@rand = rand(.quest_count);
	SilvQuest_ID = .quest_id[.@rand];
	.@monster_name$ = getmonsterinfo(.quest_mob$[.@rand], MOB_NAME);

	setquest SilvQuest_ID;

	mes "[Silvervine Quester]";
	mes "Your new assignment: ^FF0000" + .@monster_name$ + "^000000.";
	mes "Track your kill status directly inside your ^0000FFQuest Log (Alt+U)^000000.";
	close;

OnInit:
	// ==========================================
	// CONFIGURATION
	// ==========================================
	.max_daily_quests = 10;
	.reward_item = 6417; // Silvervine Fruit
	.reward_amount = 5;

	// Quest IDs must match db/re/quest_db.yml and SystemEN/OngoingQuests.lub.
	// 62928-62939 = original client Silvervine hunts; 63004-63016 = custom extensions.
	setarray .quest_id[0],
		62928, 62929, 62930, 62931, 62932, 62933, 62934, 62935, 62936, 62937, 62938, 62939,
		63004, 63005, 63006, 63007, 63008, 63009, 63010, 63011, 63012, 63013, 63014, 63015, 63016;

	setarray .quest_mob$[0],
		"ANGELING", "ARCHANGELING", "CAT_O_NINE_TAIL", "DEVILING", "DRAGON_FLY", "ECLIPSE",
		"GHOSTRING", "MASTERING", "MAYA_PUPLE", "TOAD", "VAGABOND_WOLF", "VOCAL",
		"GRYPHON", "MUTANT_DRAGON", "CHIMERA", "BLOODY_KNIGHT", "DARK_ILLUSION", "EXECUTIONER",
		"MYSTELTAINN", "TIRFING", "OWL_BARON", "A_MOBSTER", "GALION", "NECROMANCER", "G_RANDGRIS";

	.quest_count = getarraysize(.quest_id);
	end;
}
