//===== rAthena Script =======================================
//= PRO Anti-Cheat (shield.dll) Login Enforcement
//===== Description: =========================================
//= Invisible NPC that verifies shield.dll on player login and
//= periodically checks that the anti-cheat DLL is still active.
//= Runs independently of the shield_heartbeat battle flag.
//============================================================

-	script	ProAntiCheat	-1,{
	end;

OnPCLoginEvent:
	// Skip autotrade vendors and inter-server links.
	if (checkvending(strcharinfo(0)) & 2)
		end;

	@shield_checks = 0;

	.@grace = getbattleflag("shield_heartbeat_grace");
	.@interval = getbattleflag("shield_heartbeat_interval");
	if (.@grace <= 0)
		.@grace = 30000;
	if (.@interval <= 0)
		.@interval = 120000;

	// Grace is restarted on map login by the map-server; wait for it to expire.
	addtimer .@grace + .@interval, strnpcinfo(0) + "::OnShieldCheck";
	end;

OnShieldCheck:
	if (shieldactive()) {
		@shield_checks = 0;

		.@interval = getbattleflag("shield_heartbeat_interval");
		if (.@interval <= 0)
			.@interval = 120000;

		// Re-check every heartbeat interval to detect a shut-down DLL.
		addtimer .@interval, strnpcinfo(0) + "::OnShieldCheck";
		end;
	}

	@shield_checks++;

	.@max_miss = getbattleflag("shield_heartbeat_max_miss");
	.@interval = getbattleflag("shield_heartbeat_interval");
	if (.@max_miss <= 0)
		.@max_miss = 3;
	if (.@interval <= 0)
		.@interval = 120000;

	// Retry a few times before kicking, matching map-server heartbeat tolerance.
	if (@shield_checks < .@max_miss) {
		addtimer .@interval, strnpcinfo(0) + "::OnShieldCheck";
		end;
	}

	dispbottom "PRO Anti-Cheat (shield.dll) is required to play on this server.";
	kick();
	end;
}
