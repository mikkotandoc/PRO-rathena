//===== rAthena Script =======================================
//= PRO Anti-Cheat (shield.dll) - Heartbeat Enforcement
//===== Description: =========================================
//= Verifies shield.dll heartbeat responses from the map server.
//= Players without a valid heartbeat are warned and disconnected.
//============================================================

-	script	ProAntiCheat	-1,{
	end;

OnPCLoginEvent:
	if (checkvending(strcharinfo(0)) == 2)
		end;
	shieldstart();
	addtimer 60000, strnpcinfo(3) + "::OnHeartbeatCheck";
	end;

OnHeartbeatCheck:
	if (checkvending(strcharinfo(0)) == 2)
		end;
	if (!shieldactive()) {
		dispbottom "[PRO Anti-Cheat] No valid heartbeat detected. Please reconnect using the official launcher.";
		sleep2 3000;
		atcommand "@kick " + strcharinfo(0);
		end;
	}
	addtimer 15000, strnpcinfo(3) + "::OnHeartbeatCheck";
	end;
}
