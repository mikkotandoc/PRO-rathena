//===== rAthena Script =======================================
//= Legendary Hunting Missions
//===== Description: =========================================
//= Crafts cash shop weapons and armors using materials
//= farmed from level 220+ monsters via map drops.
//= Expect ~29-58 hours of material collection per item.
//============================================================

prt_mk,120,125,4	script	Legendary Hunting Missions	4_F_EDEN_MASTER,{
	if (!getbattleflag("feature.barter")) {
		mes "[Legendary Hunting Missions]";
		mes "The crafting exchange is unavailable right now.";
		mes "Please ask a GM to enable the barter system.";
		close;
	}
	mes "[Legendary Hunting Missions]";
	mes "I forge the same legendary gear sold in the cash shop,";
	mes "but only for hunters who prove themselves against";
	mes "monsters of ^0000cdlevel 220 and above^000000.";
	next;
	switch (select(
		"Craft Body Armor:Craft Footwear:Craft Garments:Craft Shields / Off-hand:Craft Accessories:Craft Headgear:Kachua Exchange:Material Guide:^777777Cancel^000000"
	)) {
	case 1:
		mes "[Legendary Hunting Missions]";
		mes "Select the body armor you wish to forge.";
		close2;
		callshop "legendary_lhm_armors";
		end;
	case 2:
		mes "[Legendary Hunting Missions]";
		mes "Select the footwear you wish to forge.";
		close2;
		callshop "legendary_lhm_footwear";
		end;
	case 3:
		mes "[Legendary Hunting Missions]";
		mes "Select the garment you wish to forge.";
		close2;
		callshop "legendary_lhm_garments";
		end;
	case 4:
		mes "[Legendary Hunting Missions]";
		mes "Select the shield or off-hand gear you wish to forge.";
		close2;
		callshop "legendary_lhm_weapons";
		end;
	case 5:
		mes "[Legendary Hunting Missions]";
		mes "Select the accessory you wish to forge.";
		close2;
		callshop "legendary_lhm_accessories";
		end;
	case 6:
		mes "[Legendary Hunting Missions]";
		mes "Select the headgear you wish to forge.";
		close2;
		callshop "legendary_lhm_headgear";
		end;
	case 7:
		mes "[Legendary Hunting Missions]";
		mes "^0055FF15 Legend Hunt Cores^000000 -> 1 " + getitemname("Cachua_Coupon");
		mes "^0055FF20 Legend Hunt Cores^000000 -> 1 " + getitemname("K_Secret_Key");
		next;
		mes "[Legendary Hunting Missions]";
		mes "^0055FF600 Legend Hunt Cores^000000 each:";
		mes getitemname("Coupon_FateSinBoots");
		mes getitemname("Coupon_GoodnEvil_Helm");
		mes getitemname("Coupon_Justice_Weapon");
		mes getitemname("Coupon_Injustice_Weapon");
		mes getitemname("aegis_1001809");
		mes getitemname("aegis_1001810");
		close2;
		callshop "legendary_lhm_kachua_ticket";
		end;
	case 8:
		mes "[Legendary Hunting Missions]";
		mes "^0055FFLegend Hunt Scrap^000000";
		mes "Common crafting fragment.";
		mes " ";
		mes "^0055FFLegend Hunt Core^000000";
		mes "Rare crafting fragment.";
		mes "MVPs drop both materials at higher rates.";
		next;
		mes "[Legendary Hunting Missions]";
		mes "These materials do ^FF0000not^000000 drop from";
		mes "monsters below level 220.";
		mes " ";
		mes "Drop rates are fixed map drops and are";
		mes "^FF0000not affected^000000 by server drop bonuses.";
		next;
		mes "[Legendary Hunting Missions]";
		mes "^777777Normal monsters (Lv 220+):^000000";
		mes "Legend Hunt Scrap up to ^0055FF2.0%^000000,";
		mes "Legend Hunt Core up to ^0055FF0.5%^000000.";
		mes "^777777MVP monsters (Lv 220+):^000000";
		mes "Scrap up to ^0055FF5.0%^000000, Core up to ^0055FF1.5%^000000.";
		close;
	case 9:
		close;
	}
	end;
}
