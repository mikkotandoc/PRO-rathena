// Roll one rare drop (rate = chance out of 10000, iRO: very rare)
function	script	F_COP_TryDrop	{
	if (rand(10000) < getarg(1))
		getitem getarg(0), 1;
	return;
}

// Warp to a Corridor floor (official entry coordinates)
function	script	F_COP_WarpFloor	{
	switch (getarg(0)) {
	case 1: warp "vis_h01", 203, 203; break;
	case 2: warp "vis_h02", 203, 203; break;
	case 3: warp "vis_h03", 203, 203; break;
	}
	return;
}

// Returns 1 if the player's pass covers the requested floor
function	script	F_COP_AllowsFloor	{
	if (cop_floor < 1) set cop_floor, 1;
	if (cop_floor == getarg(0))
		return 1;
	dispbottom "[Corridor] Your pass does not include Floor " + getarg(0) + ".";
	return 0;
}

// True if player holds an unused Corridor pass item
function	script	F_COP_HasPendingItem	{
	if (countitem(25504) >= 1) return 1;
	if (countitem(1000392) >= 1) return 1;
	if (countitem(1000393) >= 1) return 1;
	if (countitem(1000394) >= 1) return 1;
	return 0;
}

// Apply iRO phantom toughness: 100x HP (wiki), high survivability
function	script	F_COP_BuffMob	{
	.@gid = getarg(0);
	if (.@gid <= 0) return;
	getunitdata .@gid, .@ud;
	.@hp = .@ud[UMOB_MAXHP] * 100;
	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP, .@hp;
	return;
}

// Spawn one Corridor MVP and apply phantom HP scaling.
function	script	F_COP_SpawnMob	{
	monster getarg(0), 0, 0, "--ja--", getarg(1), 1, getarg(2);
	if (!$@mobid[0]) return 0;
	callfunc "F_COP_BuffMob", $@mobid[0];
	return $@mobid[0];
}

// Shared ticket validation (returns 1 if valid, 0 and ejects if not)
function	script	F_COP_HasTicket	{
	if (cop_expiry > gettimetick(2))
		return 1;
	dispbottom "[Corridor] Access denied — a valid ticket is required.";
	warp "comodo", 214, 186;
	return 0;
}

// Kill rewards per iRO wiki drop tables (all rewards are very rare except flowers)
function	script	F_COP_GiveKillRewards	{
	.@floor = getarg(0);

	// All floors — Flower Left by Phantom (Drop & MVP)
	if (rand(100) < 70)
		getitem 25503, 1;
	if (.@floor >= 2 && rand(100) < 25)
		getitem 25503, 1;
	if (.@floor == 3) {
		getitem 25503, 1;
		if (rand(100) < 35)
			getitem 25503, 1;
	}

	switch (.@floor) {
	case 1:
		callfunc "F_COP_TryDrop", 100437, 80;
		callfunc "F_COP_TryDrop", 100271, 80;
		callfunc "F_COP_TryDrop", 100023, 100;
		callfunc "F_COP_TryDrop", 100025, 100;
		callfunc "F_COP_TryDrop", 23806, 60;
		callfunc "F_COP_TryDrop", 100023, 120;
		callfunc "F_COP_TryDrop", 100025, 120;
		callfunc "F_COP_TryDrop", 23806, 80;
		break;
	case 2:
		callfunc "F_COP_TryDrop", 100271, 80;
		callfunc "F_COP_TryDrop", 100023, 100;
		callfunc "F_COP_TryDrop", 100025, 100;
		callfunc "F_COP_TryDrop", 23806, 60;
		callfunc "F_COP_TryDrop", 100023, 120;
		callfunc "F_COP_TryDrop", 100025, 120;
		callfunc "F_COP_TryDrop", 23806, 80;
		callfunc "F_COP_TryDrop", 100437, 80;
		break;
	case 3:
		callfunc "F_COP_TryDrop", 100742, 50;
		callfunc "F_COP_TryDrop", 100742, 80;
		callfunc "F_COP_TryDrop", 100023, 120;
		callfunc "F_COP_TryDrop", 100025, 120;
		callfunc "F_COP_TryDrop", 23806, 80;
		callfunc "F_COP_TryDrop", 100412, 100;
		callfunc "F_COP_TryDrop", 100413, 100;
		break;
	}
	return;
}

	mes "You will share the hunt with other adventurers there.";

		.@timeleft = cop_expiry - gettimetick(2);

	mes "All floors: ^FF9900Flower Left by Phantom^000000";
	mes "1F: Mod Cube, 9Up Box, Bravery, Wisdom, Weapon Box";
	mes "2F: 9Up Box, Bravery, Wisdom, Weapon Box, Mod Cube";

		mes "[Phantom's Spirit]";
		mes "You already hold an active Corridor pass.";
		mes "Use the portal while it lasts.";
		close;
	}
	if (callfunc("F_COP_HasPendingItem")) {
		mes "[Phantom's Spirit]";
		mes "You already hold a Corridor pass item.";
		mes "Activate it at the Corridor Portal first.";
		close;

	if (cop_cooldown > gettimetick(2)) {
		.@timeleft = cop_cooldown - gettimetick(2);

		.@hours = (.@timeleft % 86400) / 3600;
		.@mins = (.@timeleft % 3600) / 60;

		mes "The spirits are resting. You must wait:";

		else
			mes "^FF0000" + .@hours + " hour(s) and " + .@mins + " minute(s)^000000.";
		close;

		mes "Something went wrong — your seal could not be created.";
		mes "Check that you have inventory space and try again.";
		close;

	mes "[Phantom's Spirit]";
	mes "You receive a ^33AA33Phantom's Seal^000000!";
	mes "It grants ^FF00001 hour^000000 on the ^33AA331st Floor^000000.";
	mes "Take it to the Corridor Portal to begin your hunt.";
	close;

	if (cop_expiry > gettimetick(2)) {
		mes "[Phantom's Spirit]";
		mes "You already hold an active Corridor pass.";
		close;
	}
	if (callfunc("F_COP_HasPendingItem")) {
		mes "[Phantom's Spirit]";
		mes "You already hold a Corridor pass item.";
		mes "Activate it at the Corridor Portal first.";
		close;

	mes "[Phantom's Spirit]";
	mes "Trade ^FF9900Phantom's Invitations^000000 for a floor ticket.";
	mes "You hold: ^33AA33" + countitem(1000391) + "^000000 invitation(s).";

		"1F Ticket [2 Invitations]",
		"2F Ticket [3 Invitations]",
		"3F Ticket [4 Invitations]",
		"Cancel"

		.@cost = 2;
		.@ticket = 1000392;
		.@floor$ = "1st";

		.@cost = 3;
		.@ticket = 1000393;
		.@floor$ = "2nd";
		break;
	case 3:
		.@cost = 4;
		.@ticket = 1000394;
		.@floor$ = "3rd";
		break;
	case 4:
		close;

	if (countitem(1000391) < .@cost) {
		mes "[Phantom's Spirit]";
		mes "You need ^FF9900" + .@cost + " Phantom's Invitation(s)^000000.";
		mes "Obtain them from Kachua's Secret Key when available.";
		close;
	}
	delitem 1000391, .@cost;
	getitem .@ticket, 1;
	if (countitem(.@ticket) < 1) {
		getitem 1000391, .@cost;
		mes "[Phantom's Spirit]";
		mes "Something went wrong — the ticket could not be created.";
		mes "Check that you have inventory space and try again.";
		close;
	}
	mes "[Phantom's Spirit]";
	mes "You receive a ^33AA33Corridor of Phantom " + .@floor$ + " Floor Ticket^000000!";
	mes "It grants ^FF00002 hours^000000 on the ^33AA33" + .@floor$ + " Floor^000000.";
	mes "Present it at the Corridor Portal to enter.";
	close;

	mes "[Phantom's Spirit]";
	.@flowers = countitem(25503);
	if (.@flowers < 10) {
		mes "You need ^FF990010 Flowers Left by Phantom^000000.";
		mes "You currently hold: ^FF9900" + .@flowers + " flower(s)^000000.";
		close;
	}
	mes "You hold ^FF9900" + .@flowers + " Flowers^000000.";
	mes "Sacrifice ^FF990010 Flowers^000000 for a chance at a rare reward?";
	next;
	if (select("Offer the Flowers:No, keep them") == 2) close;

	if (.@roll < 400) {
		// ~4% — Sealed MVP card (iRO wiki; DB-available sealed cards)

		mes "An ^33AA33Ancient Hero weapon^000000 emerges from the veil.";
	} else {
		// ~70% — boss class card
		setarray .@boss[0],
			4135, 4168, 4137, 4121, 4146, 4142, 4143, 4318,
			4147, 4128, 4131, 4132, 4123, 4144, 4145, 4148;
		getitem .@boss[rand(16)], 1;
		mes "[Phantom's Spirit]";
		mes "A ^33AA33boss class card^000000 drifts to you from beyond.";
	}
	close;

OnKickOut:
	if (compare(strcharinfo(3), "vis_h")) {
		dispbottom "[Corridor] Your ticket has expired.";
		warp "comodo", 214, 186;
	}
	end;
}

// Floor 1 MVP Spawner -- vis_h01
// -------------------------------------------------------
vis_h01,2,2,0	script	#cop_spawner_f1	HIDDEN_WARP_NPC,{
	end;

OnInit:
	callsub S_EnsureSpawn;
	end;

OnPCLoadMapEvent:
	callsub S_EnsureSpawn;
	end;

S_EnsureSpawn:
	if (.spawned) return;
	.spawned = 1;
	callsub S_SpawnAll;
	return;

S_SpawnAll:
	.@map$ = "vis_h01";
	.@ev$ = strnpcinfo(0) + "::OnBossKill";

	// mob id, count per type
	setarray .@mobs[0],
		1038, 4,	// Osiris
		1039, 4,	// Baphomet
		1086, 3,	// Golden Thief Bug
		1087, 3,	// Orc Hero
		1112, 4,	// Drake
		1115, 3,	// Eddga
		1147, 3,	// Maya
		1150, 3,	// Moonlight Flower
		1159, 3,	// Phreeoni
		1190, 3,	// Orc Lord
		1272, 3,	// Dark Lord
		1059, 3;	// Mistress

	.@size = getarraysize(.@mobs);
	for (.@i = 0; .@i < .@size; .@i += 2) {
		for (.@n = 0; .@n < .@mobs[.@i+1]; .@n++)
			callfunc "F_COP_SpawnMob", .@map$, .@mobs[.@i], .@ev$;
	}
	return;

OnBossKill:
	callfunc "F_COP_GiveKillRewards", 1;

	if (killedrid <= 0) end;
	.@idx = getarraysize(.respawn_queue);
	.respawn_queue[.@idx] = killedrid;
	if (!.respawn_timer) {
		.respawn_timer = 1;
		stopnpctimer;
		initnpctimer;
	}
	end;

OnTimer120000:
	if (getarraysize(.respawn_queue) > 0) {
		.@mid = .respawn_queue[0];
		deletearray .respawn_queue[0], 1;
		callfunc "F_COP_SpawnMob", "vis_h01", .@mid, strnpcinfo(0) + "::OnBossKill";
	}
	if (getarraysize(.respawn_queue) > 0) {
		stopnpctimer;
		initnpctimer;
	} else {
		.respawn_timer = 0;
		stopnpctimer;
	}
	end;
}
